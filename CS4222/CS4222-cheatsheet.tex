\subsection{CS4222 Course Overview}

\subsection{Internet of Things (IoT)}

\begin{itemize}
\item \textbf{IoT}: Network of embedded devices, sense, compute, communicate. \textbf{4 Core Tasks of IoT Device}: Compute and Storage, Sense and Actuation, Power Management, Wireless Networking.
\item \textbf{Typical IoT Architecture}: Sensors and actuators at the edge (ADC: Analog to Digital Converter). Embedded processor for computation (MCU: Microcontroller Unit). Wireless interface for communication. Backend/cloud for storage and analytics.
\item \textbf{ARM}: ARM (Advanced RISC Machines): low-power RISC (reduced instruction set computer) processor architecture (and ecosystem), widely used in embedded and IoT devices due to its energy efficiency, small footprint, and suitability for battery-powered wireless systems. 
\item \textbf{ARM Holdings Ltd.} provides the Instruction Set Architecture (ISA), CPU core designs (small part of the chip), tools etc. and licenses to semiconductor companies (e.g., Qualcomm, Apple, Samsung) to design and manufacture their own ARM-based chips.
\item \textbf{Prevalence}: Billions of embedded devices, 4.4B Cortex-M processors in Q4 2020.
\item \textbf{Energy Trend}: Computation cost low, Wireless comms energy expensive.
\end{itemize}


\subsection{History of Wireless Communication}
\begin{itemize}
  \item \textbf{Early Wireless Milestones}: 1880: Photophone (sound over light), 1893: Wireless radio communication, 1899: km-scale radio transmission, 1901: Transatlantic wireless.
  \item \textbf{Broadcasting Evolution}: 1906: First AM radio audio broadcast. 1935: Frequency Modulation (FM)
  1920s--30s: Radio stations worldwide. 1980s: Digital Audio Broadcasting (DAB)
  \item \textbf{Television Broadcasting}: 1923: First wireless TV transmit. 1941: NTSC standard (US). 1962: Telstar satellite. 1967: PAL standard (Analog TV). 2000s: Digital TV transition
  \item \textbf{WiFi Timeline}: 1985: ISM band opened for unlicensed use. 1997: IEEE 802.11 standard. 1999: 802.11a/b and WiFi Alliance. 1999: First consumer WiFi (Apple iBook)
  \item Every decade, lower prices, new usage, establish new industry.
\end{itemize}

\subsection{Basics of Wireless Communication}
\begin{itemize}
\item \textbf{Radio}: Latin \textit{radius} (ray). EM waves (3 Hz to 300 GHz, collectively called \textbf{radio spectrum}).
\item Radio technology implies transmitting and receiving EM waves using of frequency in radio spectrum, also known as radio waves.
\item \textbf{Radio Transceivers}: for transmission/reception of EM waves. WiFi, Bluetooth, IoT devices
\item \textbf{Electromagnetic (EM) Waves}: EM waves consist of oscillating electric and magnetic fields. Frequency: cycles per second (Hz). Period inversely proportional to frequency.
\item EM waves travelling in free space, speed of light ($ c\approx 3 x10^8$ m/s). $\lambda = \frac{c}{f}$. ($v = fx$).
\item \textbf{E.g. Wavelengths}: 98.7 MHz (FM) $\Rightarrow$ $\lambda = 3$ m; 2.4 GHz (WiFi) $\Rightarrow$ $\lambda = 0.125m$.
\end{itemize}

\section{2. Link Budget and Antenna Design}

\subsection{Antenna Basics, Directionality, Types}
\begin{itemize}
  \item \textbf{Radio Transceiver}: Electrical device that enables transmission and reception of EM waves.
  \item \textbf{Antenna}: Electrical conductor, transmit, receive EM waves. Converts electrical signals to radio waves and vice versa, reciprocal (equal performance for transmit TX and reception RX).
  \item \textbf{Types of Directionality}: directional, omnidirectional (radiates equally in all directions).
  \item \textbf{Common Antenna types}: Monopole, Array, Parabolic. Patch (flat element on substrate), loop, aperture etc. Dipole: Two conductors in opposite directions. 
\end{itemize}

\centerline{\includegraphics[width = 1\linewidth]{lobes}}

\subsection{Antenna Size and Frequency}
Antenna size is \textbf{inversely proportional} to frequency of radio wave ($\lambda = c/f$)
\begin{itemize}
  \item Higher frequency, smaller antenna size, Lower frequency, larger size
  \item Dipole: length of conductor is $\frac{1}{2}$ of wavelength of radio wave
  \item Half-Dipole: length of conductor is $\frac{1}{4}$ of wavelength of radio wave
  \item Antennas work efficiently by resonating w. radio waves, when size is fraction of wavelength.
\end{itemize}



\subsection{Performance of Antenna: Antenna Gain}
\begin{itemize}
  \item \textbf{Gain of antenna:} Dimensionless ratio (unitless in linear form), commonly expressed in decibels (dB) or dBi. Measures ability of antenna to focus energy in a particular direction compared to a reference.
  \begin{itemize}
    \item \textbf{dBi}: Gain referenced to a theoretical isotropic antenna (uniform radiation in all directions)
    \item \textbf{dBd}: Gain referenced to a half-wave dipole antenna ($dBd = dBi - 2.15$)
    \item \textbf{Note}: Mathematically, dB and dBi treated same in a link budget, but conceptually they defer: dB represents actual loss or gains, dBi models antenna performance due to directionality.
  \end{itemize}
  \item \textbf{Units:}
  \begin{itemize}
    \item Linear gain ($G$) is unitless, converting to decibel scale to compresses ratios.
    \item Logarithmic gain ($G_{\text{dB}}$ or $G_{\text{dBi}}$) is in decibels (dB)
    \item Convert: $G_{\text{dB}} = 10 \log_{10}(G_{\text{linear}})$ and $G_{\text{linear}} = 10^{G_{\text{dB}} / 10}$
  \end{itemize}
  \item \textbf{When to convert:}
  \begin{itemize}
    \item Use \textbf{linear gain} in physical formulas (e.g., Friis, $G = \frac{4 \pi A_e}{\lambda^2}$), can convert to dBi after.
    \item Use \textbf{dBi/dB} when doing link budget calculations (additive terms with power in dBm, FSPL in dB, etc.)
  \end{itemize}
  \item $\uparrow$ gain $\Rightarrow$ more directional transmission/reception $\Rightarrow$ narrower beamwidth
\end{itemize}

\begin{multicols}{2}
  \[
    \boxed{G = \frac{4 \pi A_e}{\lambda^2} = \frac{4 \pi f^2 A_e}{c^2}}
  \]
  \columnbreak
  \begin{itemize}
    \item $G$ = antenna gain (unitless)
    \item $A_e$ = effective aperture area (m$^2$)
    \item $f$ = carrier frequency (Hz)
    \item $c$ = speed of light $3 \times 10^8$ m/s
    \item $\lambda$ = carrier wavelength (m)
  \end{itemize}
\end{multicols}

\begin{itemize}
  \item \textbf{Directional Dependence:} Both $G$ and $A_e$ are functions of direction. The value used in link budgets is typically the \textbf{peak value} (main lobe).
  \item \textbf{Effective Area ($A_e$):} Represents the "power-capturing" cross-section.
  \item Power output evaluated in a direction, relative to isotropic antenna output.
  \item \textbf{Array Gain}: $G_{\text{array}} = G_{\text{single element}}(\text{dBi}) + 10 \log_{10}(N)$, where $N$ is the number of antenna elements in array; $10 \log_{10}(N)$ is the array factor. Multiply unitless gain also works.
\end{itemize}

\subsection{Radiation Pattern and Beamwidth}
\begin{itemize}
  \item \textbf{Radiation Pattern}: graphical representation of relative strength of radiated or received power in various directions. 3D or 2D plots.
  \begin{itemize}
    \item \textbf{Main Lobe}: direction of maximum radiation/reception.
    \item \textbf{Side Lobes}: smaller lobes representing radiation/reception in other directions.
    \item \textbf{Back Lobe}: lobe opposite to main lobe.
    \item \textbf{Nulls}: directions with minimal radiation/reception.
  \end{itemize}
  \item \textbf{Beamwidth}: angular separation between the $-3,\text{dB}$ points (half-power points) of the main lobe. It defines the directivity and coverage area of the antenna.
  \begin{itemize}
    \item \textbf{$-3,\text{dB}$ points}: points on the radiation pattern where power drops to half of maximum value. ( $10\log_{10}(0.5) \approx -3,\text{dB}$).
    \item \textbf{E.g. Antennas}: parabolic (highly directional), Yagi-Uda (directional), dipole antennas.
  \end{itemize}
\end{itemize}

\subsubsection{Antenna Beamwidth Tradeoffs:}
\begin{itemize}
    \item \textbf{Narrow beamwidth}: Offers higher directivity and longer range due to increased gain, making it suitable for point-to-point links.
    \item \textbf{Broader beamwidth}: Covers a larger area, broader coverage, but has lower directivity and range due to lesser gain, making it suitable for broadcast applications.
\end{itemize}


\subsubsection{Line of Sight (LoS) vs Non-Line of Sight (NLoS) Propagation}
Waves suffer attenuation due to distance, obstacles, interference. Underwater favors sound over radio waves.
\textbf{Multipath}: signal takes two or more paths from transmitter to receiver.

\begin{itemize}
  \item \textbf{Line-of-Sight (LoS) Propagation}: clear, direct, unobstructed path. Common in open areas.
  \item \textbf{Non-Line-of-Sight (NLoS) Propagation}: obstacles, reflection, diffraction, scattering. Common in urban, indoor environments. Reduces reliability and range.
\end{itemize}

\subsection{Attenuation and Gain: Decibel (dB) and dBm Units}
\begin{itemize}
  \item \textbf{Attenuation:} Loss of energy when signal traverses through a medium
  \item \textbf{Gain:} $\uparrow$ in energy $\rightarrow$ devices employed to improve signal strength (E.g. amplifiers, antennas)
  \item \textbf{Decibels (dB)}: logarithmic unit to express ratios of power levels. We use decibels to express loss/attenuation of signal and gain. (express power ratios)
  \item \framebox{$dB = 10\log_{10} \frac{P_2}{P_1}$} (where $P_1$ is input power, $P_2$ is output power)
  \begin{itemize}
    \item e.g. signal travels through medium, power $\downarrow$ to $\frac{1}{2}$ original value, (i.e. $P2 = \frac{1}{2} P1$).\\ $\implies$ $10\log_{10}\frac{P_2}{P_1} = 10\log_{10}0.5 = 10(-0.3) = -3dB$
  \end{itemize}
  \item $-3\text{dB}$ change is lose half signal power; $+10\text{dB}$ is $10\times$ increase, $20\text{dB}$ is $100\times$.
  \item Use dB to measure changes in signal strength, as decibel values can be added and subtracted instead of multiplied and divided. E.g. signal experiences a $-3,\text{dB}$ loss followed by a $+10,\text{dB}$ gain, net change is $-3 + 10 = +7,\text{dB}$, power increase of about $5\times$.
\end{itemize}

\subsubsection{Unit of power of a Signal: dBm (Decibel milliwatt)}
\begin{itemize}
  \item Decibel milliwatt (dBm) is a measure of signal power.
  \item \framebox{$P(\text{dBm}) = 10\log_{10}(P_{\text{mW}})$} (where $P_{\text{mW}}$ is power in milliwatts)
  \item $0 \ dBm$ represents a transmit power of $1 mW \rightarrow$ common reference point in comm. systems
  \item $30 \ dBm$ corresponds to transmit power of $1 W$, often max permissible output power for devices in unlicensed frequencies bands, e.g. WiFi, Bluetooth.
  \item E.g. Calculate power of signal with $dBm = -30$. \\
    $-30 = 10 \log_{10}(P_{\text{mW}}) \implies P_{\text{mW}} = 10^{-3}\text{mW} = 0.001 \text{mW}$.
  \item Key values: $0\text{dBm} = 1\text{mW}$. $20\text{dBm} = 100\text{mW}$. $30\text{dBm} = 1\text{W}$.

\end{itemize}


\subsection{Estimating received Signal Strength: Friis propagation model}
\begin{multicols}{2}
  \begin{center}
    \[
      \boxed{P_r = G_r G_t \left(\frac{c}{4\pi f_c d}\right)^a P_t} 
    \]
    \tiny{(in \ dBm)}
  \end{center}

  \columnbreak
  \begin{itemize}
    \item $f_c$ center frequency in Hertz
    \item $c$ = speed of light
    \item $d$ = distance btw transmitter/receiver
    \item $a$ = path loss component
    \item $G$ = antenna gain
    \item $P$ = power received/transmitted
  \end{itemize}
\end{multicols}

\subsection{Wireless Signal Propagation Calculations}
\begin{itemize}
  \item \textbf{Friis Propagation Model}: Estimates received signal strength and range. Assumes LoS, free-space conditions, no obstacles or interference.
  \begin{itemize}
    \item \textbf{Friis Transmission Equation}: $P_r = G_r G_t \left( \frac{c}{4\pi f_c d} \right)^\alpha P_t$.     
    \item the term $\left( \frac{\lambda}{4\pi d} \right)^\alpha$ captures power reaching receiver, aka path loss factor. FSPL isolates and inverts this term to quantify signal attenuation (loss) over distance in free space.
  \end{itemize}
\end{itemize}



\subsection{Free Space Path Loss  $FSPL (\text{linear}) = {(\frac{4\pi d f }{c}})^2$}
\begin{itemize}
  \item Loss of power by signal as it transmits from the transmitter to receiver 
  \item $d$ is the distance between the transmitter and receiver antennas (in meters),
  \item $f$ is the frequency of signal (in Hertz), $c$ is speed of light in a vacuum
  \item \framebox{\parbox{\dimexpr\linewidth-1\fboxsep-150\fboxrule}{% 
  $FSPL = 20\log_{10}(d_{\text{m}}) + 20\log_{10}(f_{\text{Hz}})+ 20\log_{10}(\frac{4\pi}{c})$
  }} \tiny{| in $dB$ always}
\end{itemize}

\begin{itemize}
  \item Represents absolute minimum loss any wireless link will experience, real channels always worse. FSPL helps quantify the cost of moving to different frequencies and distances.
  \item \textbf{FSPL Formula Forms}:
  \begin{itemize}
    \item $\text{FSPL}(dB) = 32.44 + 20\log_{10}(d_{\text{km}}) + 20\log_{10}(f_{\text{MHz}})$ 
    \item $\text{FSPL}(dB) = 92.45 + 20\log_{10}(d_{\text{km}}) + 20\log_{10}(f_{\text{GHz}})$ (equivalent forms)
  \end{itemize}
  \item Doubling distance or frequency each on own increases loss by $6,\text{dB}$ (inverse square law).
\end{itemize}

\subsection{Total link gain in wireless communication}
\begin{itemize}
  \item Net amplification/attenuation of EM signal along the communication path.
  \\ Antennas, FSPL, any other system component is included. 
  \begin{itemize}
    \item \framebox{\parbox{\dimexpr\linewidth-1\fboxsep-150\fboxrule}{% 
        $\text{(dB)} = G_{Tx} + G_{Rx} - L_{Path} - L_{Misc}$
      }}
      \item $G_{Tx}$ = Transmitter antenna gain (dB)
      \item $G_{Rx}$ = Receiver antenna gain (dB)
      \item $L_{Path}$ = Path loss (dB), including free-space loss
      \item $L_{Misc}$ = misc losses (dB), (atmospheric/cable losses/interference)
  \end{itemize}
\end{itemize}


\subsection{Link Budget, Link Margin, Total Link Gain}
\begin{itemize}
  \item \textbf{Link Budget}: Calculate received signal strength given all gains/losses in comm. system
  \item Total Link Budget ($P_{rx}$) = $P_{tx} + $ Total Link Gain 
  \item \textbf{Total Link Budget} \framebox{\parbox{\dimexpr\linewidth-1\fboxsep-280\fboxrule}{% 
    $ P_{Rx}= P_{Tx} + G_{Tx} + G_{Rx} - L_{Total}$
  }}
  \begin{itemize}
    \item $P_{\text{Tx}}$ = Transmit power in dBm.
    \item $G_{\text{Tx}}$ = Transmit antenna gain in dBi.
    \item $G_{\text{Rx}}$ = Receive antenna gain in dBi.
    \item $L_{\text{Total}}$ = Total losses (e.g., free space path loss, cable losses) in dB.
    \item $P_{\text{Rx}}$ = Received power in dBm.
  \end{itemize}
  \item \textbf{Link Margin}: Received power $-$ receiver sensitivity (\textbf{Signal Sensitivity}). (Represents buffer excess above bare minimum needed)
  \item \textbf{Google Project Loon Case Study}: Balloons at 20km altitude vs. satellites at 400km. Lower latency/power. -100 dBm sensitivity device comms balloon 20km away, only 5.3mW power.
\end{itemize}

\subsection{dB, dBm and dBi operations}
\begin{itemize}
  \item \textbf{db} (Decibel): log unit for ratios (gain/loss), dimensionless multiplier, no abolute meaning
  \item \textbf{dBm}: absolute power reference to $1mW$, $0 dBm = 1mW$ (physical quantity)
  \item \textbf{dBi}: Antenna gain relative to isotropic radiator. (directionality)
  \item \textbf{Adding dB values:}
  \begin{itemize}
    \item If 2 components contribute gain/loss, we add their dB values 
    \item (logarithms turn multiplication into addition) $\rightarrow$ e.g. 2 amplifiers, +10dB, +5dB gain
  \end{itemize}
  \item \textbf{Adding dBi to dBm to get EIRP:}
  \begin{itemize}
    \item \textbf{Effective Isotropic Radiated Power (EIRP)}: total output power w. main lobe gain.
    \item \framebox{$EIRP (\text{dBm}) = P_{TX}(\text{dBm}) + G_{antenna}(\text{dBi}) - L_{\text{cable}}$}
    \item \textbf{Concept}: It treats the \textit{max main lobe gain} as if it were radiated \textit{omnidirectionally}.
    \item \textbf{Note}: "Theoretical" power. The actual total energy leaving the antenna is still $P_{TX}$.
  \end{itemize}
  \item \textbf{Subtracting Losses (dB) from Power (dBm)}
  \begin{itemize}
    \item Losses (cable/FPSL) can be subtracted from transmitted power
  \end{itemize}
  \item \textbf{Subtracting dBm values to get dB ratio difference:} Used to compare 2 power EIRP.\\
   (E.g. $30 dBm - 20 dBm \rightarrow 10 dB$ difference in power, note output unit is dB, not dBm)
  \item \textbf{Cannot add two dBm values directly}: Log values cannot be added directly
  \begin{itemize}
    \item Must convert to linear scale e.g. $10 dBm = 10mW$. $20 dBm = 100 mW$, Total power = $110 mW = 10\log_{10}(110) = 20.4 dBm$
  \end{itemize}
\end{itemize}

\section{3. Modulation Techniques for Wireless Communication}

\subsubsection{How are radio waves modified to carry information}
How information 'rides' on radio waves. Radio waves are pure sine waves that carry no information. To modify to send information (e.g. speech, bits):

\begin{itemize}
  \item Start with low frequency baseband signal (message signal)
  \item Use it to modify a high frequency carrier signal (radio wave)
  \item Transmit modified carrier signal over the air
  \item Receiver demodulates to recover baseband signal
\end{itemize}
This process is called \textbf{modulation}. Modulation Techniques include Amplitude Modulation (AM), Frequency Modulation (FM), Phase Modulation (PM), or vary them together and advanced digital schemes like QAM, PSK.

\subsection{Modulation}
Encoding 'baseband' information into analog (continuous, smooth waveform) 'carrier' signal. Baseband signal often $\downarrow$ frequency than carrier. Frequency of carrier dictates spectrum usage.

\begin{itemize}
  \item \textbf{Why use a high-frequency carrier instead of sending baseband directly?}
  \begin{itemize}
    \item Baseband signals or info change at a much lower rate. (E.g. Speech up to $20$KHz, temp changes $\approx$ few Hz) - these signals are slow compared to radio waves (MHz to GHz).
    \item If transmit baseband directly, need large antennas (size $\approx \frac{1}{4} \lambda$), and frequency overlaps.
  \end{itemize}

  \item \textbf{Benefits of modulating onto high frequency carrier}
  \begin{itemize}
    \item More efficient (smaller) antenna size, better radio spectrum usage. (e.g. 2.401 GHz, 2.412 GHz - frequency division multiplexing).
  \end{itemize}
\end{itemize}

\centerline{\includegraphics[width = 1\linewidth]{modulation.jpg}}
\centerline{\includegraphics[width = \linewidth]{modulations.jpg}}

\begin{itemize}
  \item \textbf{Modulation Index $\mu$, $\beta$}: Extent carrier signal is modulated w. message signal. Determines strength of modulated signal, impacts bandwidth, power distribution, overall signal quality.
\end{itemize}

\subsection{Amplitude modulation (AM)}
\begin{itemize}
  \item Change amplitude of carrier signal according to the message signal: $s(t)$: modulated signal
      \[
        \boxed{s(t) = [A_c + A_m \cdot m(t)] \cos (2\pi f_c t)}
      \]
      \begin{multicols}{2}
        \begin{itemize} 
          \item $A_c$ Carrier amplitude
          \item $A_m$ Message (baseband) amplitude
          \item $m(t)$ Message signal
          \item $f_c$ Carrier Frequency
        \end{itemize}
    \end{multicols}
  \item \textbf{Signal (Single Tone)}: $s(t) = A_c [1 + \mu \cos(2\pi f_m t)] \cos(2\pi f_c t)$
  \item \textbf{AM Modulation Index $\mu$}
  \begin{itemize}
    \item For amplitude modulation $(modulation \ index): \mu = A_m / A_c$
    \begin{itemize}
      \item $A_m$ = Peak amplitude of modulating/baseband signal
      \item $A_c$ = Peak amplitude of carrier signal
      \item For AM: $\mu < 1$: under-modulated (weak signal, no distortion); $\mu = 1$: fully modulated; $\mu > 1$: over-modulated (distortion, clipping)
    \end{itemize}
  \item \textbf{Multi-tone Modulation Index ($\mu_{total}$)}: message with multiple frequencies with individual indices $\mu_1, \mu_2, \dots, \mu_n$: $\mu_{total} = \sqrt{\mu_1^2 + \mu_2^2 + \dots + \mu_n^2}$
    % \item \textbf{Condition for Distortion:} The peak sum ($\mu_1 + \mu_2 + \dots$) must stay $\leq 1$ to avoid instantaneous overmodulation, though $\mu_{total}$ is used for power.
  \end{itemize}
  \item \textbf{Frequencies}: Carrier ($f_c$), Lower Sideband ($f_c - f_m$), Upper Sideband ($f_c + f_m$).
  \item \textbf{Bandwidth}: $\boxed{BW_{AM} = 2 \cdot f_{m,max}}$ (Twice the \textit{highest} message frequency).
\item \textbf{AM Power Analysis}
    \begin{itemize}
      \item Carrier Power: $P_c = \frac{A_c^2}{2R}$ (where $R$ is antenna resistance, often $50\Omega$)
      \item Total Transmitted Power: $\boxed{P_t = P_c \left(1 + \frac{\mu_{total}^2}{2}\right)}$
      \item Efficiency ($\eta$): $\eta = \frac{P_{sidebands}}{P_{total}} = \frac{\mu_{total}^2}{2 + \mu_{total}^2}$
    \end{itemize}

  \item \textbf{AM Spectrum and Bandwidth}
    \begin{itemize}
      \item Frequencies present: $f_c$, $f_c \pm f_m$.
      \item Bandwidth: $\boxed{BW_{AM} = 2 \cdot f_{m,max}}$ 
      \item \textit{Note: BW is always twice the highest frequency component of the message.}
      
    \end{itemize}
\end{itemize}

\subsection{Frequency modulation (FM)}
\begin{itemize}
  \item Change frequency of carrier signal according to the message signal
  \item Advantages over AM: resilient to interference, spectrum efficiency, used for radio
  \item \textbf{Bandwidth}: how much "room" signal occupies in frequency spectrum. FM bandwidth depends on how \textit{fast} and \textit{how much} you wiggle carrier freq around $f_c$ to carry information.
      \[
        \boxed{s(t) = A_c cos (2\pi f_c t + 2\pi k_f \int m(t) dt))}
      \] 
    \begin{itemize}
      \item $A_c$ Carrier amplitude, $f_c$ Carrier Frequency
      \item $k_f$ Frequency sensitivity of modulator (how much carrier freq. changes in response to amplitude of modulating signal)
      \item $m(t)$ Modulating signal
    \end{itemize}
  \item \textbf{FM Modulation Index: $\beta$}
  \begin{itemize}
    \item $\beta(modulation \ index) = \triangle f / f_m$
    \item $\triangle f$ = Peak frequency deviation (how much carrier frequency changes)
    \item $f_m$ = Maximum frequency of the modulating signal
    \item $\beta < 1$: Narrow band signal, $\beta > 1$ Wideband signal.
    \item FM broadcast uses wideband signal, better noise immunity (less static, hiss)
    \item Given by Carson's rule, bandwidth $B \approx 2 (\triangle f + f_m)$
  \end{itemize}
\end{itemize}

\subsubsection{What is a phase}
\begin{itemize}
  \item Represents position of point on waveform cycle, an offset from starting position of sine wave
  \item Complete cycle of sine wave is 360 degress / 2 $\pi$ radian
  \begin{multicols}{2}
      \[
        \boxed{s(t) = A \cos(2\pi ft + \phi)}
      \]
    \columnbreak
    \begin{itemize}
      \item $A$ = Amplitude of signal
      \item $f$ = Frequency of signal
      \item $\phi$ Phase (phase contstant / offset)
    \end{itemize}
  \end{multicols}
  \item $\phi$ determines where in its cyclewave is (begins at $t=0$)
  \item Changing $phi$ shifts waveform left or right in time
  \item \textbf{\underline{Impact of phase shifts on waves (summation)}} (interference)
  \begin{itemize}
    \item \textit{Constructive:} 2 waves with same phase add tgt to form larger amplitude
    \item \textit{Destructive:} 2 waves out-of-phase (e.g. 180$\deg$ apart), cancel each other
  \end{itemize}
\end{itemize}

\subsection{Phase modulation}
\begin{itemize}
  \item Change phase of carrier signal according to the message signal
  \item Integral part of modern communcation standards: Wifi/Television
  \item Advantages over FM: Better SNR and spectral efficiency
  \item Uses different phase shifts to represent digital data
  \begin{multicols}{2}
    \begin{center}
      \[
        \boxed{s(t) = A \cos(2\pi f_c t + 2\pi k_p m(t))}
      \]
    \end{center}
    \columnbreak
    \begin{itemize}
      \item $A$ = Amplitude of signal
      \item $f_c$ = Carrier frequency
      \item $k_p$ = Phase sensitivity (rad / unit amplitude)
      \item $m(t)$ = modulating signal
    \end{itemize}
  \end{multicols}
\end{itemize}

\subsubsection{Binary Phase Shift Keying}
\begin{itemize}
  \item simple form of phase modulation where phase is shifted between 2 values 
  \item \textit{Bit 0} $\rightarrow$ 0 degrees ($A\sin (2\pi f_c t)$)
  \item \textit{Bit 1} $\rightarrow$ 180 degrees ($A\sin(2\pi f_c t + \pi)$)
\end{itemize}
\includegraphics*[width=8.5cm, height=2.8cm]{bpsk.png}

\subsubsection{Quadrature Binary Phase Shift Keying}
\begin{itemize}
  \item extends BPSK with 4 distinct phase states, allow encoding of 2 bits/symbols
  \item \textit{Bit 00} $\rightarrow$ 45 degrees ($A\sin (2\pi f_c t + \pi/8)$)
  \item \textit{Bit 01} $\rightarrow$ 90 degrees ($A\sin(2\pi f_c t + \pi/4)$)
  \item \textit{Bit 10} $\rightarrow$ 135 degrees, \textit{Bit 11} $\rightarrow$ 225 degrees
  \item We are \textit{doubling the bit rate} for the same bandwidth
\end{itemize}

\subsubsection{Digital equivalent of AM FM and PM modulation}
\begin{itemize}
  \item Vary analog signal according to digital bits
  \begin{itemize}
    \item ASK: amplitude/FSK: frequency/PSK: phase shift keying
  \end{itemize}
  \item Digital variants relevant with respect to computer communication
\end{itemize}

\section{4. Channel Capacity and Link Quality}


\subsection{Wireless Link Quality and Reliability Metrics:}
\begin{itemize}
  \item \textbf{Packet Delivery Ratio (PDR)}: ratio of successfully received packets to transmitted packets. 

  \centerline{
    \framebox{\parbox{\dimexpr\linewidth-1\fboxsep-280\fboxrule}{% 
      $PDR = \frac{Received \ Packets}{Transmitted \ Packets} \times 100$
    }}
  }
  \begin{itemize}
    \item $\uparrow$ = better link quality, $\downarrow$ = congestion/interference or poor connectivity
  \end{itemize}

  \item \textbf{Bit Error Rate (BER):} Ratio of bits received in error to the total transmitted bits. Indicates transmission quality at the physical layer.
  
  \centerline{
  \framebox{\parbox{\dimexpr\linewidth-1\fboxsep-240\fboxrule}{% 
    $BER = \frac{Errored \ Bits}{Total \ Bits \ Transmitted} \times 100$
  }}
  }
  \begin{itemize}
    \item $\downarrow$ = fewer errors + better link perf. $\uparrow$ = data corruption / retransmissions
    \item Wired Ethernet: $10^{-12}$, Wireless Wifi: $10^{-6}$ to $10^{-3}$, Poor cxn = $10^{-2}$ 
  \end{itemize}

  \item \textbf{Signal-to-Noise Ratio (SNR):} Ratio of signal power to background noise, often measured in decibels (dB). 
  
  \centerline{
    \framebox{\parbox{\dimexpr\linewidth-1\fboxsep-250\fboxrule}{% 
      $SNR = 10\log_{10}(\frac{P_{signal}}{P_{noise}})$ | if $P$ in $W$
    }}
  }
  \begin{itemize}
    \item Else $P_{signal} - P_{noise}$ in $dBm$
    \item $\uparrow$ = stronger signal, fewer errors. $\downarrow$ = more packet loss. 
    \item SNR $> 25 dB$ = excellent (4G LTE etc). SNR $< 10 dB$ = unreliable link.
  \end{itemize}

  \item \textbf{Expected Transmission Count (ETX):} Measures expected number of trasmissions (including re-trasmission) required for packet to successfully received. Used in routing algorithms.
  
  \centerline{
    \framebox{\parbox{\dimexpr\linewidth-1\fboxsep-340\fboxrule}{% 
      $ETX = \frac{1}{P_{forward} \times P_{reverse}}$
    }} 
  }
  \begin{itemize}
    \item Lower ETX means fewer transmission and better efficiency, higher leads to increased latency and power consumption
  \end{itemize}             
\end{itemize}

\subsubsection{Understanding Channel Capacity \& Link Quality}
\begin{itemize}
    \item \textbf{Definition:} Max rate information can be transmitted over comm. channel without error.
    \item Channel capacity is measured in the unit of \textbf{bits/second}.
    \item How does channel capacity relate to link quality?
    \begin{itemize}
        \item Higher \textbf{SNR} usually leads to a higher capacity. (Bandwidth is the frequency range over which signal is transmitted; wider bandwidth more data can fit, higher rates)
        \item Factors that reduce link quality: \textbf{Fading, interference}, reducing channel capacity.
    \end{itemize}
    \item Channel Capacity and Link Quality | Shannon Perspective:
    \begin{multicols}{2}
      \begin{center}
        \[
          \boxed{C = B \times \log_2(1 + \frac{S}{N})}
        \]
      \end{center}
      \columnbreak
      \begin{itemize}
        \item $C$ Channel capacity in bits per second (bps)
        \item $B$ = Bandwidth of the channel in Hertz (Hz)
        \item $S/N$ Signal-to-noise ratio (SNR) expressed as linear ratio (convert: $10^{\text{SNR(dB)}/10}$)
      \end{itemize}
    \end{multicols}
\end{itemize}

\subsubsection{Channels, Subchannels and Overall Wireless Link Quality}
\begin{itemize}
  \item Total bandwidth divided into \textbf{subchannels}, each w. own bandwidth ($B_i$), SNR ($S/N_i$). 
  \begin{itemize}
      \item Capacity per subchannel: $C_i = B_i \log_2(1 + S/N_i)$
      \item Overall channel capacity = sum of capacities of all subchannels: $C_\text{total} = \sum_i C_i$
      \item Intuition: Some parts of the channel may experience more noise or interference than others. Treating them separately allows a more accurate capacity estimate.
  \end{itemize}
\end{itemize}

\subsubsection{Link Quality and Channel Capacity Relationship}
\begin{itemize}
    \item \textbf{Link quality} is determined by reliability (PDR, ETX), physical-layer conditions (BER, SNR).
    \item \textbf{High PDR} and \textbf{low ETX} indicate reliable, efficient packet delivery with fewer retransmissions.
    \item \textbf{Low BER} implies fewer bit errors and less need for retransmission, improving throughput.
    \item \textbf{Higher SNR} leads to fewer errors and directly increases achievable channel capacity.
    \item \textbf{Channel capacity} increases with both \textbf{bandwidth} and \textbf{SNR}.
    \item \textbf{Poor link conditions} (fading, interference, noise) reduce effective capacity even if theoretical capacity is high. In practice, \textbf{wireless systems aim to maximize SNR and minimize interference} to approach theoretical capacity limits.
\end{itemize}

\subsubsection{Sensors}
Electronic component that measures physical phenomenon.
\subsubsection{Models of sensors}
\begin{itemize}
  \item Many sensors can be modelled approximately as an affine function
  \item $f:R \rightarrow R$
  \begin{itemize}
    \item $x(t)$: Physical Quantity reported by sensor at the time instance $t$
    \item $f(x(t))$: Value reported by sensor for the particular physical quantity
    \item Function $f$ is linear if there exists proportionality constant $a \in R$ and a bias $b \in R$ s.t. $\forall x(t) \in R$, $f(x(t)) = ax(t) + b$
  \end{itemize}
  \item \textbf{Proportionality constant $a$:} represents \textit{sensitivity} of a sensor, specifies degree sensor reading changes to change in physical quantity.
  \item No sensor truly \textit{realizes an affine function}, range of sensor, set of values of a physical quantity that measures is always limited. Outside range, an affine function model is no longer valid.
  \item Physical quantities outside this range typically \textit{saturate}
  \begin{itemize}
    \item These quantities yield a max or min reading \underline{outside their range}
  \end{itemize}
  \item Embedded systems, focus on \textit{passive sensing}, much cheaper, lower energy consumption
\end{itemize}

\subsubsection{Sensors transform phenomenon into electrical signals}
\begin{itemize}
  \item \textbf{\underline{Ohm's Law}}: voltage ($V$) = Current ($I$) $\times$ Resistance ($R$)
  \item By altering one of the variables, in response to changes in the phenomenon, a sensor generates an analog signal that corresponds with the phenomenon
  \begin{itemize}
    \item e.g. Resistivity $\rho$: materials with higher $\rho$ create more resistance for same length/area. $R = \frac{\rho L}{A}$ Sensitive to physical conditions:
    \begin{itemize}
      \item Temperature: Many conductors have higher resistivity at higher temps
      \item Light: materials (photoresistors) change resistivity exposed to light
      \item Strain: e.g. strain gauges increase in resistivity when stretched
    \end{itemize}
  \end{itemize}
  \item \textbf{\textit{Temperature}} sensing using resistivity changes
  \begin{itemize}
    \item Thermistor is a sensor which varies in resistance based on temperature
    \begin{itemize}
      \item \textbf{Advantages:} cheap and easy to use , low cost
      \item \textbf{Disadvantages:} non-linear transfer function
    \end{itemize}
  \end{itemize}
  \item \textbf{\textit{Light}} sensing using resistivity changes
  \begin{itemize}
    \item Photocell change resistance with light (non-linear)
  \end{itemize}
\end{itemize}

\subsubsection{Voltage Dividers}
\begin{multicols}{2}
  \begin{itemize}
    \item $V_{out} = \frac{R_2}{R_1 + R_2} \times V_{in}$
    \begin{itemize}
      \item $V_{in}$ is a voltage source and $R_1$ and $R_2$ are resistors
    \end{itemize}
    \item If $R_1 == R_2$, $V_{out} = \frac{V_{in}}{2}$
    \item Smaller $R_1$ means larger $V_{out}$, $V_{out}$ approaches $V_{in}$
  \end{itemize}
  \includegraphics*[width=2.2cm, height=2cm]{voltagedividers.png}
\end{multicols}

\subsubsection{Analog Digital Converter (ADC): Bridging analog and digital worlds }
\begin{itemize}
  \item ADCs convert analog signals (continuous range of values) into digital signals (discrete numerical representations). Essential for digital devices to process real-world analog inputs.
    \item \textbf{What are the various parameters of the ADC?}
    \begin{itemize}
        \item \textbf{Resolution:} Determined by the number of bits the ADC uses to represent the analog value. More bits mean higher resolution and finer granularity in the output.
        \item \textbf{Sampling Rate:} Frequency ADC samples the analog signal. Higher sampling rates capture more detail, necessary to comply with Nyquist theorem. Also higher power consumption.
        \item \textbf{Accuracy:} The closeness of the ADC's output to the true input value. It includes factors such as linearity, noise, and error rate.
    \end{itemize}
\end{itemize}

\subsubsection{Digital Sensors}
\begin{itemize}
    \item Some sensors incorporate ADC within their circuits to provide digitized signals, which can then be read by a microcontroller.
    \item \textbf{Dynamic Range:} Digital sensors are unable to distinguish between two closely spaced values of the physical quantity. The precision $p$ of a sensor is the smallest absolute difference between two values of a physical quantity whose sensor readings are distinguishable.
    \begin{itemize}
      \item $D \in R+$ of a digital sensor is the ratio
      \item $p$ = represents the precision of the sensor: $D = \frac{H - L}{p}$, where $H$ and $L$ are limits of range in (often expressed as units of decibels)
      \item $D_{dB} = 20\log_{10}(\frac{H-L}{p})$, $p= \frac{H-L}{2^{ADL} - 1}$, where we take $-1$ because we usually do not use 0 to represent
    \end{itemize}
\end{itemize}

\subsubsection{Quantization of sensor readings}
\begin{itemize}
  \item Digital sensors represent a physical quantity through a $n$-bit number
  \item $n$ is a typically small number, and thus there are $2^n$ numbers
  \item Actual physical quantity represented by a real number $x(t) \in R$
  \item Digital sensor picks \textbf{one of the $2^n$} distinct numbers to represent it, thus sensor readings are quantized
\end{itemize}

\subsubsection{Ideal digital sensor what should be $n$ and $p$}
\begin{itemize}
  \item \textbf{Ideal digital sensor:} 2 physical quantities that differ by the precision $p$ are represented by digital quantites that differ by 1 bit
  \item Precision and quantization become intertwined
\end{itemize}

\subsubsection{Sensor Distortion Function}
\begin{itemize}
  \item Defines the output of a sensor as a function of input: $f:R \rightarrow \{0, 1, \cdots, 7 \}$
  \item For sensors with output similar to above, has precision given by following formula $p = \frac{H_L}{2^n}$
\end{itemize}

\subsubsection{Sampling through analog to digital converter}
\begin{itemize}
  \item Physical quantity $x(t)$ is a function of time $t$
  \item A digital sensor will sample the physical quantity at any particular points in time to create a discrete signal
  \item Uniform sampling, there is a fixed time interval $T$ between samples
  \item $T$ is called the sampling interval
  \item The resulting signal may be modelled as function $s : Z \rightarrow R$ $\forall n \in Z , s(n) = f(x(NT))$
  \item Physical quantity $x(t)$ is observed only at times $t = nT$
\end{itemize}

\subsubsection{How does this related to ADCs}
\begin{itemize}
  \item The smaller the sampling interval $T$, more costly to provide more bits in an ADC.
  \item Same cost, faster ADCs typically produce fewer bits and hence have either higher quantization error or smaller range
\end{itemize}

\subsubsection{Noise in sensors and measurement}
\begin{itemize}
  \item Part of signal which we do not want. If we want to measure $x(t)$ and we ended up measuring $x'(t)$ the noist $n(t) = x'(t) - x(t)$
  \item Noise is the side effect of sensor not measuing exactly what it is supposed to measure: e.g. by sensor imperfection, quantization errors etc
\end{itemize}

\subsection{Channelization Protocols}

Channelization: divide single communication medium into multiple distinct channels, transmit without collisions through preassign, but rigid allocation wastes capacity under bursty traffic.

\subsubsection{TDMA | Time Division Multiple Access}
\begin{itemize}
  \item Split transmission in time, devices share same channel
  \item Splits time into \underline{fixed-length} windows: each device assigned 1 or more window, can build priority system here with uneven split among devices
  \item Requires syncrhonization between devices
  \begin{itemize}
    \item Often devices must listen periodically to resynchronize
    \item Less efficient use of slots reduce synchronization
    \item Large guard windows e.g. $1.5$ second slot for $1$ second trans.
  \end{itemize}
\end{itemize}

\subsubsection{FDMA | Frequency Division Multiple Access}
\begin{itemize}
  \item Separate transmissions by placing then at diff freq. Transmit at same time. 
  \item Technically, each device can use a separate, non-overlap, fixed frequency
  \begin{itemize}
    \item E.g. walkie-talkies (mostly separate), $\uparrow$ throughput and reliability, spatial diversity
    \item \textbf{Disadvantages:} If devices not always transmitting, part of spectrum is not used, inefficient.
  \end{itemize}
  \item \textbf{WiFi} - Overlapping channels, pack more concurrent users (OFDMA).
\end{itemize}

\subsubsection{Multi Channel MAC Protocols}
\begin{itemize}
  \item Control channel: default, time syncrhonization for control/msg exchange
  \item Nodes switch to different channel for data communication
  \item All channels used for comm. , node switches channel when link quality bad
\end{itemize}

\subsubsection{Orthogonal Frequency Division Multiplexing | OFDM}
\begin{multicols}{2}
  \begin{itemize}
    \item FDM: entire frequency spectrum is dedicated to carry data from a single source

    \item OFDM: data divided into many (slower) bit streams and multiple subcarriers (each with smaller spectrum used)

\subsection{Random Access Protocols}
In Random Access Protocols (RAPs), users are allowed to transmit at any time without pre-scheduling, but risk collisions. 

For dynamic wireless networks, fixed scheduling often inefficient or impractical:
\begin{itemize}
  \item Pre-allocate transmit slots inefficient when traffic is bursty/unpredictable, waste resources if nothing sent during allocated slots.
  \item RAPs are efficient, lightweight for managing sporadic,
  unscheduled, low-rate data transmits.
\end{itemize}


\subsection{Aloha}
\begin{itemize}
  \item Devices transmit whenever data is available, without explicit coordination, no prior scheduling or channel sensing; purely random transmissions 
  \item \textbf{Rules:}
  \begin{enumerate}
    \item If you have data, transmit immediately
    \item After transmission, wait a certain time to receive acknowledgment (ACK)
    \item If ACK is received, transmission successful; if no ACK, wait a random time interval and retransmit (exponential backoff recommended) 
  \end{enumerate}
  \item Collision Problem:
  \begin{itemize}
    \item If $\geq 2$ nodes transmit simultaneously, collision occurs, results in lost data
    \item Nodes do not listen before transmitting (no channel sensing)
  \end{itemize}

  \centerline{\includegraphics[width = 0.8\linewidth]{packetcollision1.jpg}}

  \item \textbf{Slotted Aloha:}
  \begin{itemize}
    \item Slotted ALOHA reduces collision windows by aligning transmissions to predefined slots
    \item Requires sync: devices must know slot boundaries, add complexity to implementation
    \item Slotted throughput is double as “before” collisions can no longer occur

  \centerline{\includegraphics[width = 0.8\linewidth]{packetcollision2.jpg}}
  \end{itemize}

  \item \textbf{ALOHA Throughput}: Shown that traffic maxes out at ALOHA throughput of 18.4\% (slotted ALOHA doubles to 36.8\%), assuming Poisson distribution of traffic.
\end{itemize}

\subsubsection{Why Use Random Access if Efficiency is Limited}
\begin{itemize}
  \item \textit{Simplicity and low overhead:}
  \begin{itemize}
    \item RAPs do not need complex synchronization or scheduling mechanisms
    \item Suitable for resource-constrained IoT or sensor networks
  \end{itemize}
  \item \textit{Efficiency in low-load scenarios:}
  \begin{itemize}
    \item Collisions are rare when network load is low
    \item Random access can achieve low latency for intermittent or bursty transmissions
  \end{itemize}
  \item \textit{Real-world usage examples:}
  \begin{itemize}
    \item BLE beaconing: Short, random broadcasts to discover devices
    \item LoRaWAN: Simplicity and battery efficiency preferred over maximizing throughput
    \item Satellite and RFID networks: Sporadic data transmission scenarios
  \end{itemize}
\end{itemize}

\subsection{Carrier Sense Multiple Access w/ Collision Avoidance (CSMA/CA)}
\textbf{CSMA/CA} medium access control protocol widely used in wireless networks (especially WiFi). Nodes sense medium to check if busy before attempting transmission, collisions actively avoided rather than detected after occurrence.

\begin{itemize}
  \item Wireless networks have difficulty reliably detecting collisions directly due to signal
  fading, interference, and hidden nodes.
  \item Collision Avoidance proactively reduces the probability of collisions occurring.
\end{itemize}


\subsubsection{CSMA/CA Algorithm Steps:}
\begin{enumerate}
  \item \textit{Listen first (Carrier Sensing):} Nodes listen before transmission. If channel busy, node waits.
  \item \textit{Random Backoff (Collision Avoidance):} Once channel idle, nodes don't transmit immediately. Wait a random time interval (random backoff period) before transmitting.
  \item \textit{Transmit Data:} After random wait, if medium remains idle, node transmits its data.
  \item \textit{Acknowledgment (ACK):} Successful reception is confirmed through an ACK from the receiving node. Lack of ACK implies potential collision or error, triggering retransmission.
\end{enumerate}

\subsubsection{Choosing $W_{max}$: Maximum Backoff Window Size}
\begin{itemize}
  \item Intuitively, $W_{max}$ should be small when lower chance of collision
  and vice versa
  \item $W_{max}$ varies according to the Binary Exponential Backoff (BEB) algorithm.
  \item Set “slot time” to 2 times the max. propagation delay on the channel
  \item If this is the first transmission, sends immediately
  \item If there collision, after i-th collision, pick $W_{max}$ randomly between $0$ and $2^i -1$ time slots.
  \item \textbf{Optimal point}: depends on no. of nodes condenting + traffic load. Too aggressive (small $W_{max}$): more collisions, too conservative (large $W_{max}$): underutilization of channel.
\end{itemize}

\subsubsection{Binary Exponential BBackoff (BEB)}
\begin{itemize}
  \item If collisions occur, wait intervals adaptively increase to reduce future collision probability
  \item Each successive collision indicates higher channel contention
  \begin{enumerate}
    \item Initial attempt: Small waiting period (randomized within a small range).
    \item After each collision: The range (window size) for the random backoff doubles (exponential increase), reducing collisions progressively:
    \begin{itemize}
      \item 1st collision: Wait between $0 - (2^1-1)$ slots
      \item 2nd collision: Wait between $0 - (2^2-1)$ slots
      \item After i-th collision: Wait between $0 - (2^i-1)$ slots
    \end{itemize}
  \end{enumerate}
\end{itemize}

\subsection{TCSMA with RTS/CTS (Threshold CSMA)}
\begin{itemize}
  \item \textbf{Hidden Terminal Problem}: Occurs when two transmitters cannot sense each others transmissions but interfere at a common receiver. (Not hearing another sender does not necessarily mean the receiver is free). RTS/CTS Mechanism:
  \begin{itemize}
    \item \textbf{Request to Send (RTS):} Short control frame sent by transmitter requesting access
    \item \textbf{Clear to Send (CTS):} Receiver replies w. CTS if ready to one node at a time.
    \begin{itemize}
        \item \textbf{Deferring}: Nearby nodes hearing CTS for a different receiver address (RA) must defer for time specified in Duration field.
        \item \textbf{Backoff (BEB)}: If RTS fails (no CTS received), sender assumes a collision and performs a Binary Exponential Backoff before retrying.
    \end{itemize}
    \item \textbf{Upside}: RTS collisions are faster, less wasteful than hidden terminal collisions
    \item \textbf{Downside:} overhead is high for waiting for CTS when contention is low + network overhead
  \end{itemize}
\end{itemize}

\section{8. Medium Access Control (MAC) for Internet of Things}
IoT MAC has different goals from generic wireless MAC: energy conservation as primary objective, can tolerate low throughput and higher latency. Hence, implication that protocols for WiFi (CSMA/CA) waste a lot of energy, and IoT has purpose built MAC protocols to minimize radio-on time.

\begin{itemize}
  \item \textbf{MAC Goals for Wireless} (E.g. WiFi): General wireless MAC protocols optimize for:
  \begin{itemize}
    \item Low Latency, High Channel Utilization, Fairness (equal chance to communicate)
  \end{itemize}
  \item \textbf{MAC Goals for Low-Power IoT}: Optimize for unique requirements:
  \begin{itemize}
    \item Extremely low power consumption, low, intermittent channel utilization
    \item Energy conservation becomes the primary design goal, MAC protocols must minimize power usage to prolong battery life, tradeoff latency or reduced throughput.
  \end{itemize}
\end{itemize}
\smallskip

\centerline{\includegraphics[width = 0.95\linewidth]{macLifecycle.jpg}}

\begin{itemize}
  \item \textbf{Idle listening} consumes significant energy, especially at low load, can consume between $50-100\%$ of energy needed for actual receiving. But need to listen for incoming transmissions, so cannot simply turn off radio.
  \item \textbf{Duty cycled MAC protocol}: periodically wake up briefly to sense transmissions/channel activity, return to sleep mode. Nodes sleep when inactive, minimize idle listening time.
  \item \textbf{Challenges}: Determining the optimal wake-up schedule, coordination to ensure nodes wake at the right time to detect transmissions.
  \item All low power MAC protocols are fundamentally about deciding when to turn the radio on and off. Trade-off latency, energy.
  \item \textbf{Duty Cycle Formula}: $\text{Duty Cycle} = \left( \frac{t_{\text{active}}}{t_{\text{active}} + t_{\text{sleep}}} \right) \times 100\%$
\end{itemize}

\subsection{Design Space of Low-Power MAC}
\begin{itemize}
  \item \textbf{Synchronous vs Asynchronous}: Does network share common schedule?
  \item \textbf{Sender-initiated vs. Receiver-initiated}: who starts handshake?
  \item \textbf{Preamble-based vs. Schedule based}: How does sender reach sleeping receiver?
\end{itemize}

\subsection{8.1 Asynchronous MAC (Sender-Initiated / LPL)}
Nodes operate on independent schedules. The burden of "finding" receiver on sender.

\subsubsection{Theory: ALOHA with Preamble Sampling}
\begin{itemize}
  \item \textbf{Steps}: Sender transmits long preamble to “wake up” receiver. Receiver periodically wakes briefly to sense for preambles. Once preamble detected, receiver stays awake, receives the actual data, acknowledges.

  \centerline{\includegraphics[width = 0.8\linewidth]{preambleSampling.jpg}}
  \item \textbf{Detection}: preamble duration ($T_p \geq T_S$) must be greater or equal to receiver's sleep interval. Else, receiver may miss preamble, sleep before data starts.
  \item \textbf{Advantages:} Reduces idle listening significantly, saving power, simple to implement async
  \item \textbf{Disadvantages:} Preamble wastes energy at sender and slightly increases latency
\end{itemize}

\subsection{B-MAC: Asynchronous Medium Access Control}
Berkeley MAC protocol, use preamble sampling. Improvement over ALOHA preamble: Adds \textbf{Clear Channel Assessment (CCA)} and a configurable interface for duty cycling.

\begin{itemize}
  \item \textbf{Preamble sampling concept}: Nodes independently choose wake-up intervals (asynchronous).
  \item Operational Steps:
  \begin{enumerate}
    \item Sender transmits a long preamble
    \item Receivers periodically sample the channel
    \item If a preamble is detected, the receiver stays awake for data transmission
  \end{enumerate}
  \item \textbf{Channel sampling interval}: Determines energy usage (Receiver waking rarely saves energy, but need longer preamble, sender wastes energy, higher latency, channel blocked longer). Long preamble as duration must match sampling interval to ensure detection.
  \item Pros: Energy-efficient, simple, asynchronous operation
  \item Cons: Long preamble introduces latency and unnecessary overhead
\end{itemize}

\subsection{X-MAC: Explicit MAC (B-MAC improvements)}
\begin{itemize}
  \item Replace long continuous preamble w. \textbf{short probe packets with gaps}:
  \begin{enumerate}
    \item Sender transmits short probe packets containing the destination address
    \item Between probes, sender listens briefly for response
    \item  Target receiver wakes up, checks address, sends early ACK
    \item Sender immediately transmits the data packet
  \end{enumerate}
  \item \textbf{Early ACK}: sender stops probing as soon as target responds. On average, half preamble duration. Also \textbf{reduced latency}, data transfer begins on response, no wait for full preamble.
  \item \textbf{Non-target receivers}: sleep immediately after read address. In B-MAC all neighbours must listen to all preambles and data.
  \item \textbf{Gains}: X-MAC achieves 2-3x energy savings over B-MAC by reduce preamble by 50\% (on average target wakes up halfway) and eliminate non-target overhearing.
  \item \textbf{Limitation}: X-MAC still uses async wake-ups, sender must probe for up to entire preamble duration in worst case. Synchronous approaches (S-MAC) avoid this.
\end{itemize}
  
\subsection{ContikiMAC}
\begin{itemize}
  \item \textbf{ContikiMAC} treats data as preamble, takes X-MAC further.
  \item  No separate preamble or probe packets. Sender repeatedly transmits actual data packet until ACK is received. Data serves as wake-up signal. Receiver detects data, decodes, and ACKs.
  \item \textbf{Phase-Lock optimization}: After first successful exchange, sender learns when the receiver wakes up. Subsequent transmissions are timed to
coincide with the receiver wake-up. Reduces preamble to near-zero for known neighbors.
  \item No wasted probe packets, phase lock eliminates preable overhead for known neighbours, energy efficient in practice.
\end{itemize}
\centerline{\includegraphics[width = 0.8\linewidth]{contikimac.png}}

\subsection{8.2 Synchronous MAC (Scheduled)}

Nodes agree on a common wake-up schedule. Neighbors follow same schedule.

\subsection{S-MAC: Synchronous Duty-Cycled MAC}
\begin{itemize}
  \item Nodes agree to wake up simultaneously at predetermined intervals
  \item Each interval split into active (listen period) and inactive (sleep period)
  \begin{enumerate}
    \item 10$\%$ duty cycle: 10$\%$ of the time nodes listen; 90$\%$ sleep.
    \item For example: nodes listen for 100 ms, then sleep for 900 ms.
  \end{enumerate}
  \item \textbf{Advantages:} Predictable and coordinated, easier to manage communication during listening, lower sender energy than B-MAC.
  \item \textbf{Disadvantages:} Requires synchronization overhead (clock drift), less flexible compared to asynchronous schemes (X-MAC, B-MAC). Latency for event directly after active period. 
\end{itemize}

\centerline{\includegraphics[width = 0.8\linewidth]{syncmac.jpg}}

\subsection{T-MAC: Timeout - Improve S-MAC Efficiency}
\begin{itemize}
  \item \textbf{S-MAC Limitation}: Fixed active period length. Nodes waste energy on idle channel period.
  \item \textbf{T-MAC Improvement}: Use \textbf{adaptive active period}. Nodes stay awake only for activity.
  \item \textbf{Mechanism (Timeout $T_A$)}: If no start-of-frame is detected within $T_A$, node sleeps early.
  \item \textbf{Gains}: Under 10\% load, T-MAC can save up to \textbf{5$\times$} more energy than S-MAC.
  \item \textbf{Trade-off (Early Sleep Problem)}: A node might sleep before a neighbor has a chance to transmit to it. This is partially mitigated by using \textit{Future Request-to-Send (FRTS)}.
\end{itemize}

\subsection{TSCH: Time-Slotted Channel Hopping (802.15.4e)}
\textbf{Standard}: Part of IEEE 802.15.4e. Used in industrial IoT (WirelessHART, 6TiSCH).
\begin{itemize}
  \item \textbf{Concept}: Combine \textbf{TDMA} (time slots) w. \textbf{Frequency Hopping} (packets change channels).
  \item \textbf{Slotframe}: Time is divided into repeating slotframes. Each slot (typically 10ms) provides enough time for TX + RX + ACK + Guard time.
  \item \textbf{The Cell}: A communication link is assigned a specific (Time Slot, Channel Offset).
  \item \textbf{Key Advantages}:\textbf{Reliability}: Channel hopping avoids persistent interference (e.g., Wi-Fi on specific freq), \textbf{Determinism}: Guaranteed latency bounds as no contention within assigned cells, \textbf{Scalability}: Multiple pairs can communicate in same time slot using different channels.
\end{itemize}

\subsection{8.3 Other MAC protocols}
\subsubsection{Wake-Up Receivers (WuRx)}
\begin{itemize}
  \item \textbf{Add a second, ultra-low-power receiver}: listens continuously while \textbf{Main Radio} stays in deep sleep (5--20mA vs. 1--100$\mu$A).
  \item \textbf{Operation}: The WuRx detects a specific wake-up signal and triggers an interrupt to "power on" the main radio for data transfer.
  \item \textbf{Advantages}: Eliminates \textit{idle listening} energy on the main radio. Near-zero power consumption with almost zero latency (no waiting for duty cycle wake-up).
  \item \textbf{Challenges}: \textbf{Low sensitivity} ($\sim$$-55$dBm) means much shorter range than main radio ($\sim$$-100$dBm), \textbf{False Wake-ups}: Interference can trigger main radio, waste power.
\end{itemize}



\subsubsection{Sender-Initiated vs. Receiver-Initiated MAC (who triggers handshake)}
\begin{itemize}
  \item \textbf{Sender-Initiated}: Sender transmits preambles/probes until the receiver wakes up. (E.g. B-MAC, X-MAC, ContikiMAC.) \textbf{Trade-off}: Simple logic, but long preambles waste sender energy and increase collision probability.
  \item \textbf{Receiver-Initiated}: Receiver wakes up and broadcasts a short \textbf{beacon} ("I'm awake, send data now"). Senders listen for this beacon before transmitting. (E.g. RI-MAC, A-MAC.). \textbf{Tradeoffs}: Eliminate long preambles; more efficient for multiple senders. But, sender must stay awake (idle listen) to catch the beacon; beacons can collide.
\end{itemize}

\subsection{8.4 Beyond Collision Avoidance: Exploiting Interference}

\subsection{Capture Effect (Decoding)}
\begin{itemize}
  \item \textbf{Definition}: The ability of a receiver to successfully decode the \textbf{strongest} signal during a collision, effectively "ignoring" weaker interferers.
  \item \textbf{The SIR Rule}: Typically requires a Signal-to-Interference Ratio (SIR) $>3$--$6$dB.
  \item \textbf{Quantitative Evidence}: A signal at $-70.89$dBm (strong) can achieve 99\% reception despite collisions if it has a $\sim$10dB advantage over the weak signal.
  \item \textbf{Impact}: Provides a natural \textit{spatial priority} for closer nodes and increases actual ALOHA throughput beyond traditional theoretical limits ($Ge^{-2G}$).
\end{itemize}


\subsection{Synchronous Transmissions (Time-based)}
\begin{itemize}
  \item \textbf{Concept}: Multiple nodes transmit the \textbf{exact same packet} at the \textbf{exact same time}.
  \item \textbf{The 500ns Rule ($\Delta$)}: R can receive packet with high probability if $\triangle$ is small. For IEEE 802.15.4, if timing offset between signals is $\Delta < 500$ns (approx. 1/32 of a symbol duration), they arrive in phase.
  \item \textbf{Constructive Interference}: Instead of collision corruption, the signals reinforce each other. 
  \item \textbf{Benefit}: More transmitters: higher signal energy, higher probability of reception at receiver.
\end{itemize}

\centerline{\includegraphics[width = 0.8\linewidth]{synctrans.png}}



\subsection{Network Layer Application: Glossy Flooding}

\begin{itemize}
  \item \textbf{Traditional Simple Flooding Mechanism}: Nodes rebroadcast received packet once with duplicate detection.
  \item \textbf{Challenges}: \textbf{Broadcast Storm}: $O(n)$ transmissions, massive redundancy and energy waste. \textbf{Collision Cascade}: Nodes at same hop distance rebroadcast at similar times, making collisions nearly certain without random delays. \textbf{Latency}: Accumulates due to random backoffs (jitter). A 10-hop flood can exceed $150$ms.
\end{itemize}

\subsubsection{Solution: Glossy (Sync-TX Network Flooding)}
\textbf{Paradigm Shift}: Stop avoiding collisions; exploit them as a beneficial feature.
\begin{itemize}
  \item \textbf{Mechanism}: Receivers use hardware interrupts to retransmit a packet \textit{immediately} upon reception to meet the $<500$ns synchronization requirement.
  \item \textbf{Performance}: \textbf{Speed}: Floods a 200-node network in $\sim$5ms (vs. 100+ ms for traditional flooding). \textbf{Reliability}: high delivery rate from deterministic constructive interference.
  \item \textbf{Applications}: Sub-microsec time sync, firmware updates, \textbf{Low-Power Wireless Bus (LWB)}.
\end{itemize}

\subsection{8.5 Li-Fi Light Fidelity (802.11bb)}
\begin{itemize}
  \item \textbf{Definition}: Wireless communication using visible light (VLC) instead of radio frequencies. Standardized as \textbf{IEEE 802.11bb}.
  \item \textbf{Transmitter}: Modulate LED intensity at speeds imperceptible to the human eye. \textbf{Receiver}: A photodetector or camera decodes the light pulses back into digital data.
  \item \textbf{Dual Use}: Same LED infrastructure provides room illumination and data transmission.
  \item \textbf{Advantages}: \textbf{Massive Bandwidth}: Entire visible light spectrum ($\sim$300--800THz), orders of magnitude wider than entire RF spectrum. \textbf{Security}: Light does not penetrate walls, no eavesdropping. \textbf{RF-Safe}: No electromagnetic interference, good for hospitals, aircraft, underwater environments where RF attenuates quickly. \textbf{License-Free}.
  \item \textbf{Disadvantages}: Require \textbf{Line-of-Sight (LoS)} and direct or near-direct alignment. \textbf{Interference}: ambient sunlight or other light sources create a high noise floor. \textbf{Uplink Difficulty}: While ceiling LEDs easily provide \textit{downlink}, an \textit{uplink} (device to ceiling) often requires additional IR LEDs or hybrid RF solutions.
  \item \textbf{Typical Applications}: \textbf{Indoor Positioning}: LED ID beacons for precise location tracking, \textbf{D2D (Device-to-Device)}: Short-range data transfer (e.g., screen-to-camera), \textbf{Hazardous Environments}: Safe for use in petrochemical plants or mines where RF sparks are a concern.
\end{itemize}

\section{9. Wireless Localization}
\begin{itemize}
  \item Localization systems determine the physical coordinates of an electronic device such as IoT nodes, drones, mobile phones or a subject. 
  \textbf{Types of Target Physical Coordinates}:
  \begin{itemize}
    \item \textbf{Global (absolute) coordinates:} defined using an external reference system
    (e.g., GPS), meaning coordinates are absolute and universally understood.
    \begin{itemize}
      \item Global coordinates: (37.7749° N, 122.4194° W)
    \end{itemize}
    \item \textbf{Relative coordinates} defined with respect to another coordinate system,
    using transformation like rotation, reflection, or translation
    \begin{itemize}
      \item E.g., locating an AirTag indoors, for example, inside a room
    \end{itemize}
  \end{itemize}
\end{itemize}

\subsection{9.1 GPS | Global Positioning System}
\begin{itemize}
  \item Satellites placed in Medium Earth Orbit (Altitude 20,000 KM)
  \item Orbit twice a day, $\geq$ 4 satellites visible from anywhere on Earth at all times
  \item Steps of GPS positioning:
  \begin{enumerate}
    \item GPS satellites broadcast their \textcolor{red}{positions and time}
    \begin{itemize}
      \item Position: Global coordinates of the satellite
      \item Time: A precise timestamp of when the signal was sent (\textcolor{red}{Time of Transmission})
    \end{itemize}
    \item The GPS receiver (e.g., phone) receives signals from \textcolor{red}{multiple} satellites
    \begin{itemize}
      \item It logs the exact timestamp (\textcolor{red}{Time of Arrival}) when each signal arrives
    \end{itemize}
    \item The GPS receiver \textcolor{red}{calculates the distance} from each satellite
    \begin{itemize}
      \item \textcolor{teal}{Time of flight} = Time of Arrival - Time of Transmission
      \item \textcolor{red}{Distance} = \textcolor{teal}{time of flight} $\times$ speed of light ($3.0 \times 10^8$ m/s)
    \end{itemize}
    \item Using \textcolor{red}{trilateration}, your device computes your exact location
    \begin{itemize}
      \item With dist. measurements from each beacon, we determine the position
      \item \textcolor{red}{Goal: find intersection point} of circles(2D) / spheres (3D)
    \end{itemize}
  \end{enumerate}
  \item GPS requires at least 4 satellites
  \begin{itemize}
    \item 3D scenario: with 3 satellites, trilateration results in two possible locations. If target device on ground, the incorrect point can be easily eliminated.
    \item \textbf{\textit{However}}: Not all targets are on the ground, e.g., aircraft. Additionally, may be clock (time) drift in GPS receivers, e.g., mobile phone
  \end{itemize}
\end{itemize}

\subsubsection{GPS receiver (e.g. phone): Determining distance}
\begin{multicols}{2}
  \begin{itemize}
    \item Each satellite periodically sends a unique pseudo-random no. sequence
    \begin{itemize}
      \item Receivers also \textcolor{red}{locally generate the same sequence in synchronized fashion}
    \end{itemize}
    \item Receivers measure \textcolor{red}{time of arrival} of the sequence, use offset in sequence to calculate  \textcolor{red}{Time of Flight} and distance.
    \item Transmission also includes \textcolor{red}{time of transmission} of sequence, location of satellite at time
  \end{itemize}
  \columnbreak
  \includegraphics*[width=4cm, height=2.5cm]{determiningdistance.png}
\end{multicols}
\begin{itemize}
  \item Satellites have very accurate (expensive) clocks, and all satellites are precisely synchronized.
  \item Receivers have less accurate clock, e.g., mobile phone, much cheaper. The 4th satellite helps solve for the receiver's clock error.
  \item Overall, solves for 4 unknowns: \texttt{latitude, longitude, altitude, clock drift}
\end{itemize}

\subsubsection{Physical layer of GPS}
\begin{itemize}
  \item Frequency: $1.2$GHz and $1.5$GHz, $10-15$MHz. BPSK modulation.
  \item Receiver needs to know additional info: Current time, position of each satellite
  \item GPS transmission has this data layered on top (50bps)
  \begin{itemize}
    \item \textbf{Ephemeris}: valid for up to 4 hours. Listen (up to) 30 seconds gets time, satellite position.
    \item \textbf{Almanac}, valid for up to two weeks. Listen for 12.5 minutes gets all satellites positions.
  \end{itemize}
\end{itemize}

\subsubsection{Assisted GPS}
\begin{itemize}
  \item Cell phone GPS is so quick by downloading \textbf{almanac} from the internet
  \item Bootstraps location information, cell towers give coarse position, enables devices to know which satellites are overhead
\end{itemize}

\subsubsection{Beacon Nodes (e.g., GPS satellites)}
\begin{itemize}
  \item Necessary to localize a network in a global coordinate system, also they can be used for finding relative coordinates. At a minimum:
  \begin{itemize}
    \item Three (non-collinear) beacons required to define a global coordinate system in 2D.
    \item Four non-coplanar beacons for 3-D coordinates
  \end{itemize}
  \item \textcolor{red}{Beacon Placement}: Has significant impact on localization
  \begin{itemize}
    \item If the beacons are spread out around the target area instead of being clustered together, we get more accurate positioning
    \item Adding an extra beacon somewhere centrally can also improve results
  \end{itemize}
\end{itemize}

\subsection{9.2 Ways to measure distances}
\subsubsection{Time of Flight (ToF) / Time of Arrival (ToA)}
\begin{itemize}
  \item \textbf{Mechanism}: Determine distance by knowing exact position of infrastructure, Transmit time, Receive time, Signal velocity (e.g. light)
  \item Infrastructure transmits (can happen constantly), device can listen only when needed.
  \item \textbf{Conditions}: Requires time synchronization between infrastructure and device (Synchronization must be good: $1\mu$ s $= 300$ meters). Additional considerations: Speed of signal propagation, Distance to be measured, Clock accuracy
  \item E.g.: Speed of light ($c$) $\approx 3\times10^8$ m/s, Distance travel in 1ms = 300m, Clock accuracy needed for meter accuracy (1/$c$) $\approx 3$ns
\end{itemize}

\subsubsection{Time difference of arrival | (TDoA)}
\begin{itemize}
  \item \textcolor{red}{(Target) Device transmits} and infrastructure \textcolor{red}{(beacon) listens} transmission. Multiple connected infrastructure nodes receive at different times based on distance.
  \item \textbf{Mechanism}: Determine distance by knowing exact position of infrastructure, Time of arrival at two different locations, Signal velocity (i.e. speed of light).
  \item \textcolor{red}{Does not require synchronization with infrastructure!}
  \begin{itemize}
    \item Still requires synchronization between infrastructure nodes
    \item Does require device to transmit "loud" enough for infrastructure to hear it
  \end{itemize}
  \item With different mediums:
  \begin{multicols}{2}
    \begin{itemize}
      \item TDoA: $\frac{d}{S_{sound}} - \frac{d}{S_{Radio}} + t_{delay}$, Works well in LOS environment
      \item Requires additional hardware but works better with calibration
    \end{itemize}
    \columnbreak
    \includegraphics*[width=4cm, height=1.5cm]{tdoa.png}
  \end{multicols}
\end{itemize}

\subsubsection{Ultrasound}
\begin{itemize}
  \item \textbf{Advantages}: Solves barrier problem (cannot pass through solid objects like walls). Human spaces already designed to contain sound. Easier to get high-accuracy: Sound much slower than light. Less synchronization needed for same accuracy.
  \item \textbf{Disadvantages}: More energy to transmit, Slower update rate (still sub-second), Limited range, Pets can hear it
\end{itemize}


\subsubsection{Received Signal Strength Indicators (RSSI)}
\begin{itemize}
  \item RSSI measures \textcolor{red}{signal strength} to estimate distance. In theory, \textcolor{red}{RSSI varies with 1/d$^\alpha$}
  \begin{itemize}
    \item $\alpha$ depends on the wireless environment (typically 2 or 3)
    \item Common assumption: closer nodes receive stronger RSSI than those further away
  \end{itemize}
  \item In practice, RSSI range measurements contain noise, several meters off, highly non-uniform
  \begin{itemize}
    \item Radio propagation differs over different surface (water, asphalt etc.), obstacles (wall, furniture etc.). \textbf{Problem:} Pathloss is NOT only due to distance
  \end{itemize}
\end{itemize}

\subsubsection{Fingerprinting using RSSI}
\begin{itemize}
  \item Shifts from trilateration to mapping: Use RSSI patterns of known locations.
  \item \textcolor{red}{Phase 1 - Setup}: Measure signal strength from multiple Access Points at various locations. Record measurement in a database - mapping signal strengths to specific locations.
  \item \textcolor{red}{Phase 2 - Localization}: Device measures real-time signal strength from surrounding APs. Compare readings to the database to determine the most likely location.
  \item \textbf{Wi-Fi Fingerprinting}: Repurpose existing infrastructure for localization (e.g. Wifi points). Enable localization with unmodified hardware, no extra infra. (E.g. WiFi RADAR)
  \item \textbf{Challenges}: Efforts to create database: manual measurements at locations, Non-static environment: signal strength changes with obstacles, periodically re-measure and update database, Measurements might vary between devices (how you hold, antennas of devices etc).
  \item \textbf{Variations:} Measurements can use several access points simultaneously to improve accuracy. No need Wi-Fi based, can use cellular networks or deploy own BLE beacons
  \item \textbf{Accuracy:} State-of-the-art: median accuracy of 0.5-1.5 meters. (Enough to pinpoint room etc.) Barrier problems depend on walls.
\end{itemize}

\subsubsection{Angle of arrival (AoA)}
\begin{itemize}
  \item Measures \textcolor{red}{direction of incoming signal} using antenna array
  \item Using 2 anchors (B1, B2), $A$ can determine position. BLE 5.1 includes AoA localization
  \item Leverage antenna arrays, measure differences in arrival time ($\triangle t$) across antenna elements 
  \item $d$ (antenna spacing), $\triangle t \times \text{speed of light} = d\sin\theta$; $\implies$ $\sin \theta = \text{speed of light} \times \frac{\triangle t}{d}$
\end{itemize}

\section{10. Wireless Sensor Networks}

\subsection{10.1 Neighbour Discovery}
Once deployed, sensors have to initiate \textcolor{red}{neighbour discovery} to connect with one another.

\begin{itemize}
  \item \textbf{Naive approach}: \textcolor{red}{Continously listens} for neighbour to transmit signals until hears transmission. To wake up and constantly listen, battery drained \textbf{quickly}. Not energy efficient.
\end{itemize}

\subsection{Synchronous Neighbour Discovery}
\begin{itemize}
  \item \textbf{All nodes synchronized} to same clock/time. Nodes scheduled to wake up periodically at same time to transmit/receive.
  \item \textcolor{red}{deterministic} approach | provide guarantee bounds on discovery latency
  \item \textbf{Disadvantages}: \textcolor{red}{Synchronization} overhead, consumes significant power. \textcolor{red}{Slow discovery process} to detect all nodes. \textcolor{red}{Scalability} issues as network grows, more nodes to sync.
\end{itemize}

\subsection{Asynchronous Neighbour Discovery}
\begin{itemize}
  \item \textbf{No clock synchronization needed}: Each device uses own clock to divide timeline into time slots. Independently wakeup for TX/RX at random time slots.
  \item No restrict wake-up schedules $\rightarrow$ much lower average discovery latency
\end{itemize}

\subsubsection{Birthday Protocol - Asynchronous neighbor discovery}
Easy to implement + Good average discovery latency but \underline{unbounded worst case}
\centerline{\includegraphics[width = 0.7\linewidth]{birthdayparadox.png}}
\begin{enumerate}
  \item Nodes have repeating periods. E.g. each period has 10 time slots.
  \item Node randomly pick wake-up slots (shaded) (e.g., 1 slot per period (10\% duty cycle))
\end{enumerate}

\subsubsection{Model probability of birthday paradox}
\begin{itemize}
  \item Two nodes randomly choose to wakeup $k$ time slots (\textcolor{red}{shaded boxes}) out of a \textcolor{red}{total $n$ slots (periods)}. What is the probability that they wakeup at the same time slot at least once?
  \item Probability ($Q$) can be written down as:
  \begin{itemize}
    \item (\textcolor{blue}{Duty-cycle: 1\%}) $n = 100$, $k = 1$, $Q(100,1) = 0.01$
    \item (\textcolor{blue}{Duty-cycle: 5\%}) $n = 100$, $k = 5$, $Q(100,5) = 0.23$
    \item (\textcolor{blue}{Duty-cycle: 1\%}) $n = 1000$, $k = 10$, $Q(1000,10) = 0.096$
    \item (\textcolor{blue}{Duty-cycle: 5\%}) $n = 1000$, $k = 50$, $Q(1000,50) = 0.928$
  \end{itemize}
  \item $Q(n,k) = 1 - \frac{\binom{n-k}{k}}{\binom{n}{k}}$
  \item \textcolor{blue}{Drawbacks}: probabilistic bound for discovery time, some nodes may take very long
\end{itemize}

\subsubsection{DISCO - Asynchronous deterministic neighbor discovery}
\begin{itemize}
  \item Scheduling wake-up timeslots at multiples of prime numbers to \textcolor{red}{ensure deterministic discovery}. Nodes can have different duty cycles.
  \begin{itemize}
    \item Each node picks \textcolor{red}{two prime numbers $p_{i1}$ and $p_{i2}$}
    \item Node \textbf{wakes up every $p_{i1}$ and $p_{i2}$ time slots} to $\uparrow$ discovery chances
    \item The duty cycle is $(1/p_{i1} + 1/p_{i2})$
    \item If both nodes pick the same pair of prime numbers:
    \begin{itemize}
      \item Duty cycle $\approx (p_{i1} + p_{i2})/ p_{i1}p_{i2}$, Worst Case time $\approx p_{i1}p_{i2}$
    \end{itemize}
  \end{itemize}
\end{itemize}

\subsection{10.2 Routing}
If all N nodes connected directly, $N(N-1)$ or $O(N^2)$ uni-directional links

\subsubsection{Structure of wired networks}
\begin{itemize}
  \item \textcolor{red}{Hierarchical and structured}: \textit{switches}, \textit{routers} for managing connections and route data.
  \begin{itemize}
    \item Switch connects multiple devices within a local network (e.g., at home or in an office)
    \item Router connects different networks together (e.g., your home network to the Internet)
  \end{itemize}
  \item Add more switches/routers reduces \# of direct links needed and \textcolor{red}{allows network scaling}.
\end{itemize}

\subsubsection{Structure of wireless networks}
\begin{itemize}
  \item Wireless network are typically \textcolor{red}{single-hop:} All stations/devices \textcolor{red}{communicate directly} with base stations or (Access Points) (E.g.  WiFi, Cellular Network, LoRa)
  \item \textbf{Multi-hop}
  \begin{itemize}
    \item Extends network reach/coverage for the same number of base stations
    \item Provides more routes, potentially more robust
  \end{itemize}
\end{itemize}

\subsection{Routing Metrics: Factors to consider in Routing Decisions}
\begin{itemize}
  \item \textcolor{red}{Energy efficiency}
  \item \textcolor{red}{Reliability:} ensure packets successfully delivered (e.g., ETX)
  \item \textcolor{red}{Hop Count}
  \item Delay: how long it takes for data to reach the destination
  \item Other factors: network congestion / throughput
\end{itemize}

\subsection{Energy Consumption}
\begin{itemize}
  \item Free space model: $P_r = C P_t / d^\alpha$
  \begin{itemize}
    \item $C$ is a constant | $\alpha$ is an exponent, typically 2 or 3
  \end{itemize}
  \item Transmission over a total distance $d$, minimum received power $P_0$
  \item \textbf{Single-hop} transmission to cover distance $d$, calculate transmission power ($P_1$):
  \begin{itemize}
    \item $P_0 = CP_1/d^\alpha$ or $P_1 = P_0/C * d^\alpha$
  \end{itemize}
  \item \textbf{Multiple-hop} ($N$) transmissions to cover distance $d$, calc transmission power ($P_N$) per hop:
  \begin{itemize}
    \item $P_0 = CP_N/(d/N)^\alpha$ or $P_N = P_0/C * (d/N)^\alpha$
    \item Total transmission power is $N P_N = N P_0/C * (d/N)^\alpha$
  \end{itemize}
\end{itemize}

\subsection{Reliability}
\begin{itemize}
  \item Ensure packets successfully delivered, minimize packet loss
  \item Routing Metric: \textcolor{red}{``expected number of transmissions''} to successfully transmit a packet (ETX)
  \item Measure \textcolor{red}{link quality} over time to determine each link's reliability
  \begin{itemize}
    \item Link quality varies between nodes
    \item Fewest hops might not be the ``most reliable'' path
  \end{itemize}
\end{itemize}

\subsubsection{Measure Packet Reception Ratio (PRR)}
\begin{itemize}
  \item A transmitter sends \textcolor{red}{$S$ packets} over a period of \textcolor{red}{$T$ seconds}. The \textcolor{red}{receiver} records number of packets successfully received, denoted as $R$. $PRR = R/S$
  \item \textbf{Overheads of Measuring PRR}: Trade-off communication cost and reaction time.
    \begin{itemize}
    \item Assume ZigBee radio is used (250Kbps), each transmission takes 1ms ($\sim$250bits, $\sim$30bytes) and link measurement probes should not collide
    \item If T=1s, S=100, Measurement takes up 10\% of bandwidth 1 node. If there are 10 nodes, 100 probes need to be transmitted in 1 sec (100\% of bandwidth used only for measurement!)
    \item If T=100s, S=10, Measurement takes up very little bandwidth, but it takes a long time to get an accurate estimate of link quality. If link quality changes during that time, the measurement may be inaccurate. Good S and T values application dependent.
    \end{itemize}
\end{itemize}

\subsubsection{Calculate Expected number of Transmissions: ETX}
\begin{itemize}
  \item ETX (expected transmissions) is a \textcolor{red}{commonly used routing metric}
  \item Using Packet reception ratio (PRR): $ETX = 1/PRR$
\end{itemize}

\subsection{Routing Algorithms}

\subsubsection{Route using shortest path algorithm}
\begin{itemize}
  \item Many practical routing algorithms based on notion of shortest path between two nodes. Each link assigned a positive number called its length. Route each packet along minimum length (or shortest) path between source and destination.
  \item If length is always 1, shortest path becomes minimum hop routing
  \item By defining the length using metrics like ETX, energy, delay etc, other paths can be obtained
\end{itemize}

\subsection{Classification of Routing Protocols}
\begin{itemize}
  \item \textbf{Proactive (Routing Table based):} when a packet needs to be forwarded, the \textcolor{red}{best route is already known (stored in the routing table)}
  \item \textbf{Reactive (Demand-based):} Determine the best route only when there is data to send
\end{itemize}

\subsubsection{Proactive Routing}
\begin{itemize}
  \item \textcolor{red}{Periodically} query for the possible routes in the network. Save all routes that are important (or all routes), also determine routes whenever topology changes (nodes join/leave)
  \item \textbf{Upside:} when a packet arrives, route to destination is already known
  \item \textbf{Downside:} requires more memory and effort on part of routers. Wastes some network bandwidth on checking for route changes
\end{itemize}

\subsubsection{Reactive Routing}
\begin{itemize}
  \item Opportunistically build up a map of the routes through network, hopefully optimal routes.
  \item Map routes in reaction to a packet arrival. 
  \item \textbf{Rationale}: Sensor devices are slow and limited, most likely to resend to same prior address. Discover a route when it is needed, then cache for next time
\end{itemize}

\subsubsection{Ad-hoc Networks}
\begin{itemize}
  \item \textbf{A Mobile Ad hoc Network (MANET)} is an autonomous system of mobile (non-static) nodes (MS) (also serving as routers) connected by wireless links. No infrastructure exists.
  \item Network's wireless topology may change dynamically in an unpredictable manner as nodes are free to move, each node has limited transmitting power.
  \item \textbf{Assumption}: Due to node mobility, routing path may be available, needs to be ``discovered''
\end{itemize}

\subsection{Ad-hoc On-demand Distance Vector Routing (AODV)}
\begin{itemize}
  \item \textbf{Ad-hoc:} dynamic and temporary networks
  \item \textbf{On-Demand}: Routes are only created when a source node has data to send.
  \item \textbf{Distance vector:} routing metric is \textcolor{red}{the number of hops} a packet has to pass (\textcolor{red}{Hop Count})
\end{itemize}

\subsubsection{AODV Routing Table}
Each node maintains a table that stores:
\begin{itemize}
  \item \textbf{Destination IP Address}: The final target.
  \item \textbf{Next Hop}: \textcolor{red}{Stores only the immediate neighbor} towards the destination, not the full path.
  \item \textbf{Hop Count}: Distance to the destination.
  \item \textbf{Destination Sequence Number}: An indicator of route \textbf{freshness}.
  \item \textbf{Lifetime}: Expiration time of the route.
\end{itemize}

\subsubsection{The 2-Step Route Discovery Process}

\textbf{1. Route Request (RREQ) -- Discovery (Broadcast)}
\begin{itemize}
  \item \textbf{Action}: When route isn't cached, source \textbf{floods} an RREQ (nodes forward it, incrementing hop count), till reaches destination or node with valid route to destination.
  \item \textbf{Reverse Path}: Each intermediate node records which neighbor sent the RREQ, creating a path back to the source.
  \item \textbf{AODV Route Requests (RREQ) Packet Structure}:
  \begin{center}
    \begin{tabular}{|c|c|c|c|c|c|}
    \hline
    Request & Source & Destination & Source & Destination & \cellcolor{orange!60}Hop \\
    ID & Address & Address & SeqNo & SeqNo & \cellcolor{orange!60}Count \\
    \hline
    \end{tabular}
  \end{center}
  \item \textbf{Sequence Numbers}: Monotonically increasing values used to determine if information is ``fresh.'' A node only updates its table if the incoming SeqNo is higher than its stored value.
  \begin{itemize}
    \item \textbf{Source SeqNo}: E.g. like a broadcast number, used by receivers to learn the "freshness" of the path \textit{back to the source}.
    \item \textbf{Destination SeqNo}: Used by the source to specify how "fresh" the route \textit{to the destination} must be that it will accept. If source does not know destination's SeqNo, it can set it to zero, meaning it will accept any route.
    \item \textbf{The Rule of Freshness}: A node updates its routing table only if:
      \begin{enumerate}
          \item $\text{Incoming SeqNo} > \text{Stored SeqNo}$ (Newer information), or (Shorter path):
          \item $\text{Incoming SeqNo} = \text{Stored SeqNo}$ \textbf{AND} $\text{Incoming Hop Count} < \text{Stored Hop Count}$
      \end{enumerate}
  \end{itemize}
  \item \textbf{Request ID}: Unique ID for RREQ to detect and discard duplicate broadcasts.
  \item \textbf{Hop Count}: The number of hops request has taken. Starts at 1, incremented by nodes.
\end{itemize}

\smallskip

\textbf{2. Route Response (RREP) -- Establishment (Unicast)}
\begin{itemize}
  \item \textbf{Action}: The destination (or a node with a valid route) replies with a unicast RREP.
  \item \textbf{AODV Route Response (RREP) Packet Structure}:
  \begin{center}
      \begin{tabular}{|c|c|c|c|}
      \hline
      Source Addr & Destination Addr & Destination SeqNo & Hop Count \\
      \hline
      \end{tabular}
  \end{center}
  \item \textbf{Forward Path}: The RREP follows the reverse path back to the source, establishing the forward route for data in the routing tables. Includes destination sequence number as a sense of recency. Note dest. addr. still refers to the original target node, same as the RREQ.
  \item \textbf{Logic Check}: The \textbf{Destination Addr} and \textbf{Dest SeqNo} still refer to the \textbf{Target Node}. The RREP updates the Dest. SeqNo before sending RREP to prove required recency.
  \item \textbf{Generating the RREP}: 
    \begin{itemize}
      \item \textbf{By Destination Node}: Increments its sequence number to ensure it is strictly greater than the one in the RREQ. This forces the network to accept the new route.
      \item \textbf{By Intermediate Node}: Can only reply if it has a route with a $Dest SeqNo \geq$ the one in the RREQ. It \textit{cannot} increment the value; it simply relays the cached $DestSeqNo$.
    \end{itemize}
\end{itemize}
  

\subsubsection{Route Maintenance (Self-Healing)}
\begin{itemize}
  \item \textbf{Link Break}: Detected when a neighbor fails to acknowledge a transmission.
  \item \textbf{RERR (Route Error)}: Node that detects break sends a RERR to all "upstream" neighbors.
  \item \textbf{Recovery}: Source node receives the RERR, deletes the stale route, and starts a new RREQ if it still needs to send data.
\end{itemize}

\subsubsection{Tradeoffs for Reactive Routing}
\begin{itemize}
  \item \textbf{Upside: Efficiency} 
  \begin{itemize}
    \item \textcolor{red}{No transmissions unless demand}: Save battery, bandwidth, avoid periodic "heartbeats".
    \item \textbf{Dynamic Tolerance}: Routes can break/change w/o penalty if no data currently being sent.
  \end{itemize}
  
  \item \textbf{Downside: Performance Costs}
  \begin{itemize}
    \item \textcolor{red}{Large, Variable Delay}: The first packet must wait for the full RREQ/RREP "handshake" to complete across the network before it can be sent.
    \item \textbf{Discovery Overhead}: In high-mobility or high-traffic networks, frequent RREQ broadcasts can lead to a "Broadcast Storm," clogging the channel.
    \item \textbf{Stale Cache Issues}: If a cached route is broken, the sender won't know until the data fails and a \textbf{RERR} is received, forcing a costly discovery restart.
  \end{itemize}
\end{itemize}

\subsection{In-network data aggregation}
\begin{itemize}
  \item To mitigate cost of forwarding, \textcolor{red}{compute relevant statistics along the way:} mean, max, min, median etc. (e.g. Node C sends mean of (A, B, C), instead of three separate messages).

  \end{itemize}
  \item \textbf{Applications:}
  \begin{itemize}
    \item Especially useful in a query-based data collection system
    \item Queries regard a known subset of nodes
    \item Aggregation function can be specified
  \end{itemize}
\end{itemize}

\section{11. WiFi (802.11)}
\begin{itemize}
  \item \textbf{Purpose}: High-speed, short-range (10--30m) indoor communication.
  \item \textbf{History}: FCC opened ISM bands (1985); Wi-Fi Alliance (1999), certify compatibility.
\end{itemize}

\subsubsection{WiFi Network Architecture}
\begin{itemize}
  \item \textbf{Star Topology}: All client devices connect to a central \textbf{Access Point (AP)}.
  \item \textbf{Basic Service Set (BSS)}: The combination of one AP and its connected clients.
  \item \textbf{SSID (Service Set ID)}: The network name, broadcast by the AP in \textbf{Beacons} so clients can find, identify and join the network.
\end{itemize}

\subsection{11.1 MAC Layer: How WiFi Avoids Collisions}
WiFi uses \textbf{CSMA/CA} (Carrier Sense Multiple Access with Collision Avoidance) as wireless radios cannot usually detect collisions while transmitting. \textbf{CSMA/CA Process:}
\begin{enumerate}
  \item \textbf{Carrier Sense}: Listen to the medium. 
  \item \textbf{If Free}: Wait for a specific \textbf{IFS} (Inter-Frame Spacing). If still free, \textbf{Transmit}.
  \item \textbf{If Busy (Random Backoff)}: 
  \begin{itemize}
    \item Select a random number of \textbf{Backoff Slots}.
    \item \textbf{Count Down}: Only decrement slots while the medium is idle.
    \item \textbf{On Expiry}: If the medium is busy when the counter reaches zero, \textbf{increase} the maximum backoff range (Binary Exponential Backoff) and repeat.
  \end{itemize}
\end{enumerate}

\textbf{Slot Time}: The basic unit of time for the protocol, calculated as:
\begin{center}
    $\text{Slot Time} = \text{Rx-to-Tx switch time} + \text{Processing delay} + \text{Propagation delay}$
\end{center}

\subsection{11.2 Traffic Priority - Tiered Contention Multiple Access (TCMA)}
WiFi uses different IFS (inter-frame spacing) durations to ensure certain traffic classes / packets (like ACKs) can `cut the line' before new data starts. (Prioritize communication)

\begin{itemize}
  \item \textbf{Hierarchy}: \textbf{SIFS $<$ PIFS $<$ DIFS}
  \item \textbf{SIFS (Short IFS)}: Used for \textbf{Acknowledgements (ACKs)}. Because SIFS is the shortest, an ACK will always be sent before any other device can start a new data transmission.
  \item \textbf{DIFS (Distributed IFS)}: Used for standard data packets. This is the "normal" waiting time.
  \item \textbf{Logic}: Shorter wait time = Higher priority. (PIFS: Point Coordination IFS, for control frames)
\end{itemize}

\subsection{11.3 Evolution of WiFi}
In 20 years, WiFi has evolved from 2 Mbps to 9.6 Gbps. Achieved through:
\begin{enumerate}
    \item \textbf{Capable Modulation}: Shifting from Spread Spectrum to OFDM/OFDMA.
    \item \textbf{Bandwidth Expansion}: From 2.4 GHz to 5 GHz/6 GHz, wider channels (20 to 160 MHz).
\end{enumerate}
\smallskip

\subsection{11.4 Spread Spectrum (Legacy WiFi)}
The goal of Spread Spectrum is to use a \textbf{much wider bandwidth} than required for the data rate to gain robustness against interference and noise.

\subsubsection{1. Frequency Hopping (FHSS)}
\begin{itemize}
  \item \textbf{Mechanism}: The transmitter "hops" between 80 narrow channels (1 MHz each) in a pseudo-random sequence known to both sender and receiver.
  \item \textbf{Pros}: Simple hardware; resists interference on specific frequencies.
  \item \textbf{Cons}: \textcolor{red}{Slow data rates} due to switching delays (`settling time' of radio during a hop).
\end{itemize}

\centerline{\includegraphics[width = 0.8\linewidth]{fhss.png}}


\subsubsection{2. Direct Sequence (DSSS)}
\begin{itemize}
  \item \textbf{Mechanism}: Each data bit is multiplied by a higher-rate \textbf{Chipping Code}. This `smears' the bit's energy across a wide frequency band (22 MHz in 802.11b).
  \item \textbf{Robustness}: Even if noise destroys part of the signal, the original bit can often be recovered from the remaining `chips.' (resistant to narrowband interference). (Both RX and TX knows chipping code.) Allow multiple users to share band using different codes.
  \item \textbf{Advantage over FHSS}: \textcolor{red}{Higher Data Rates} because transmission is continuous; no time is wasted hopping between frequencies.
  \item Essentially, each bit is \textcolor{red}{multiplied by a sequence of bits} with a much higher data rate. Data Bit ($t_b$) $\times$ Chipping Code ($t_c$) = Transmitted Signal. As $t_c \ll t_b$, bandwidth increases.
\end{itemize}

\centerline{\includegraphics[width = 0.9\linewidth]{dsss.jpg}}

\subsubsection{Spread Spectrum Summary}
\begin{itemize}
  \item \textbf{Key Idea}: Traffic from other users appears as low-level background noise as they use different hop sequences or chipping codes. Spreading signal over wide spectrum (FHSS: multi-freq hop or DSSS: single frequency), more resilient to interference and eavesdropping.
  \item \textbf{Redundancy}: Extra bandwidth provides frequency-domain redundancy.
  \item \textbf{Legacy Tech}: Used in 802.11 (Original) and 802.11b; also the foundation for GPS and 3G (CDMA) cellular. Each pair of the TX and RX must coordinate so RX can apply the same code to decode the data.
\end{itemize}

\subsubsection{802.11b Channels (2.4 GHz)}
\begin{itemize}
    \item \textbf{Width}: 22 MHz per channel (14 total), \textbf{Spacing}: 5 MHz spacing between channel centers.
    \item \textbf{Overlap}: Most overlap significantly, only \textbf{Channels 1, 6, and 11} considered non-overlapping.
\end{itemize}

\subsection{11.5 OFDM: Orthogonal Frequency Division Multiplexing}

\textbf{The Problem: Multipath Interference}: High-speed wireless communication faces two physical challenges due to signals reflecting off walls and objects, creating multiple paths.

\begin{itemize}
  \item \textbf{Frequency Selective Fading}: Certain frequencies in a wide band get `cancelled out' by reflections while others pass through. Significant at \textcolor{red}{high data rates} and on \textcolor{red}{wide-band channels}.
  \item \textbf{Inter-Symbol Interference (ISI)}: At high data rates, bits sent so quickly (short symbol duration), reflected `echo' of previous bit crashes into the current bit, causing distortion.
\end{itemize}

\textbf{Solution: Multi-Carrier Modulation (FDM)}: Instead of sending one very fast stream of data on a single carrier, we distribute bits over \textbf{N subcarriers} in parallel.

\begin{itemize}
  \item \textbf{Subcarriers}: Each is a narrow-band signal with its own center frequency.
  \item \textbf{Symbol Duration}: Since each subcarrier only handles $1/N$ of data, can send $N$ times \textbf{slower}.
  \item \textbf{Result}: Longer symbol durations make the signal much more resistant to ISI.
\end{itemize}

\centerline{\includegraphics[width = 0.6\linewidth]{interSymbolInterference.jpg}}

\subsubsection{The Mechanics of OFDM}

\subsubsection{Concept 1: Multi-Carrier Modulation (FDM)}
The core idea is to distribute bits over $N$ narrow subcarriers instead of one wide one.
\begin{itemize}
  \item \textbf{Symbol Duration Extension}: If you split the data across 52 subcarriers, each subcarrier only handles a fraction of the data. This allows each "symbol" to last \textbf{N times longer} in time.
  \item \textbf{Why this works}: Longer symbols are "slower." Reflections (Multipath) finish bouncing around before next symbol starts, effectively \textbf{eliminating Inter-Symbol Interference (ISI)}.
  \item \textbf{Narrow-band Advantage}: Each subcarrier is so narrow that it experiences `flat fading' (easier to fix) rather than `selective fading' (which destroys wide signals).
\end{itemize}
\centerline{\includegraphics[width = 1\linewidth]{FDM.jpg}}
\medskip

---


\subsubsection{Concept 2: The `Orthogonal' Advantage (Achieve higher data rate)}
How do we pack subcarriers densely without them bleeding into each other?
\begin{itemize}
  \item \textbf{Traditional FDM}: Requires "Guard Bands" (wasted empty space) between channels.
  \item \textbf{OFDM}: Uses \textbf{Orthogonality}. The subcarriers are mathematically spaced so that the \textbf{peak power} of one carrier happens exactly at the \textbf{null (zero) point} of all other carriers.
  \item \textbf{Spectral Efficiency}: This allows subcarriers to overlap significantly, removing the need for guard bands and maximizing the data rate.
\end{itemize}

\centerline{\includegraphics[width = 0.6\linewidth]{orthogonalSubcarriers.jpg}}


\subsubsection{Concept 3: Implementation \& 802.11a}
The math required to turn bits into these overlapping waves is complex:
\begin{itemize}
  \item \textbf{Signal Processing}: \texttt{IFFT} (Inverse Fast Fourier Transform): Used at the Transmitter to combine all subcarriers into one signal. \texttt{FFT} (Fast Fourier Transform): Used at the Receiver to split the signal back into individual subcarriers.
  \item \textbf{802.11a (First WiFi with OFDM) Specs}:
    \begin{itemize}
      \item \textbf{Band}: 5 GHz (cleaner airwaves, more channels than 2.4 GHz). \textbf{Speed}: 54 Mbps.
      \item \textbf{Modulation}: \textbf{QAM} (Quadrature Amplitude Modulation) to pack more bits per subcarrier.
    \end{itemize}
\end{itemize}

\subsubsection{OFDM vs DSSS, Trade-offs}
\begin{itemize}
  \item \textbf{Evolution:} Replace DSSS with Orthogonal Frequency Division Multiplexing
  \item \textbf{DSSS (Legacy)}: High redundancy, low speed. One signal "spread" thin.
  \item \textbf{OFDM (Modern)}: High complexity, high speed (throughput), spectral efficiency (dense subcarrier packing). Also robust against narrowband interference, handles multipath effects. Many signals "stacked" efficiently.
\end{itemize}




\subsection{11.6 MIMO: Multiple Input Multiple Output}

\textbf{Concept: Why use Multiple Antennas?} For wired, add more wires for more speed. In wireless, use multiple antennas to create "virtual wires" in the air.

\begin{itemize}
  \item \textbf{Architecture}: $N$ transmit antennas, $M$ receive antennas, create $N \times M$ subchannels.
  \item \textbf{Spatial Multiplexing (Speed)}: Sending \textit{different} data streams simultaneously across antennas. This provides a huge boost in throughput without needing more bandwidth.
  \item \textbf{Antenna Diversity (Reliability)}: Sending and receiving \textcolor{red}{multiple copies} of the \textit{same} signal. If one path is blocked by a wall, another antenna likely has a clear line of sight.
\end{itemize}

\subsubsection{Advanced Signal Processing}
\begin{itemize}
  \item \textbf{Interference Recovery}: While signals overlap in the air, the receiver uses complex math to "unmix" the streams based on the unique spatial signature of each path.
  \item \textbf{Beamforming}: Instead of broadcasting in all directions, the AP uses the interaction between antennas to \textbf{focus energy} specifically toward the client.
  \item \textbf{Benefit}: Higher Signal-to-Noise Ratio (SNR) and better range.
\end{itemize}



\subsubsection{802.11n (WiFi 4): Use MIMO}
2009: 802.11n standard transformed WiFi from single-lane to multi-lane.

\begin{itemize}
  \item \textbf{Wider Channels}: Doubled bandwidth from 20 MHz to \textbf{40 MHz}. (Channel bonding: combine two adjacent channels into one big channel)
  \item \textbf{Dual-Band}: The first standard to work on both \textbf{2.4 GHz} and \textbf{5 GHz} bands.
  \item \textbf{Throughput}: Boosted maximum rates up to \textbf{600 Mbps} (up from 54 Mbps in 802.11g).
  \item \textbf{Legacy}: 802.11n remains the "baseline" for 2.4 GHz today, as many later standards (like 802.11ac) focus strictly on the 5 GHz band.
\end{itemize}

\subsubsection{Multi-User MIMO (MU-MIMO)}
Introduced with 802.11ac, solve `crowded room' problem, many devices connected to one AP.

\begin{itemize}
  \item \textbf{Single-User (SU-MIMO)}: AP talks one device at a time. Other devices wait (Sequential).
  \item \textbf{Multi-User (MU-MIMO)}: AP uses beamforming to talk to \textbf{multiple devices simultaneously} on the same frequency (Parallel).
  \item \textbf{802.11ac Limitation}: Supports \textbf{Downlink only} (AP to devices). Devices still contend for the medium when uploading to the AP.
\end{itemize}

\centerline{\includegraphics[width = 0.9\linewidth]{multiUserMIMO.jpg}}

\subsubsection{802.11ac - Enhanced MIMO and wider channels}
\begin{itemize}
  \item \textbf{Implementation (2013):} Update for \textcolor{red}{5 GHz band only}, supports Downlink MU-MIMO (from AP to device), channel widths up to \textcolor{red}{160 MHz}, and higher modulation (256-QAM).
  \item \textbf{Deployment Strategy:}: \textcolor{red}{Routers apply 802.11ac to 5 GHz and 802.11n to 2.4 GHz}
\end{itemize}

\subsection{11.7 WiFi 6 (802.11ax): High Efficiency Wireless}

\textbf{Goal}: Improve total throughput across \textit{all} devices in a crowded network (High Efficiency), rather than just the peak speed of one device.

\subsection{OFDMA (Orthogonal Frequency Division Multiple Access)}
\begin{itemize}
  \item \textbf{From OFDM to OFDMA}: turn OFDM into access control mechanism. More efficient use of bandwidth in high-density environments.
  \begin{itemize}
    \item \textbf{OFDM}: One user takes the \textit{entire} channel for a duration of time, even if they only have a small packet to send (wasted bandwidth).
    \item \textbf{OFDMA}: The channel is divided into \textbf{Resource Units (RUs)}. The AP can allocate different subcarriers to different users \textit{simultaneously}. Multiple users can transmit simultaneously, reduced latency for small data transfers.
  \end{itemize}
\end{itemize}

\centerline{\includegraphics[width = 0.7\linewidth]{ofdma.png}}


\subsubsection{802.11ax (WiFi 6) - 6 GHz Band Extension and Features}
\begin{itemize}
  \item \textbf{Standard Timeline:} Approved in 2021, first devices started supporting it in 2019 (WiFi 6).
  \item \textbf{Spectrum}: Works in 2.4 GHz, 5 GHz, and the new \textbf{6 GHz} band (WiFi 6E).
  \item \textbf{6 GHz Band}: Provides 1.2 GHz of new, unlicensed bandwidth (huge increase channels).
  \item \textbf{Up/Down MU-MIMO}: Unlike WiFi 5, support Multi-User MIMO, both uploads/downloads.
  \item \textbf{1024-QAM}: Packs 10 bits per symbol (25\% increase over 256-QAM) for short-range speed.
\end{itemize}


\subsubsection{WiFi Evolution Summary}
\begin{center}
    \begin{tabular}{|l|c|c|l|}
    \hline
    \textbf{Standard} & \textbf{Bandwidth} & \textbf{Max Rate} & \textbf{Key Technology} \\ \hline
    802.11g & 20 MHz & 54 Mbps & OFDM \\ \hline
    802.11n (WiFi 4) & 40 MHz & 600 Mbps & MIMO + Channel Bonding \\ \hline
    802.11ac (WiFi 5) & 160 MHz & 3.46 Gbps & DL MU-MIMO + 256-QAM \\ \hline
    802.11ax (WiFi 6) & 160 MHz & 9.6 Gbps & OFDMA + 1024-QAM + 6 GHz \\ \hline
    \end{tabular}
\end{center}
\begin{itemize}
    \item \textbf{Legacy (11b/g)}: Basic wireless connection.
    \item \textbf{Middle (11n/ac)}: Increasing the speed for a single user (MIMO, Wider Channels).
    \item \textbf{Modern (11ax)}: Managing high-density environments with many IoT and mobile devices (OFDMA, Scheduling).
\end{itemize}

\section{12. Low-Power IoT}

IoT devices: power, sensors, wireless communication. Categorize by operating scale (range).

\subsubsection{Categories of IoT Wireless Networks}
\begin{itemize}
    \item \textbf{PAN (Personal Area Network)}: Range $< 30$ m. E.g.: \textbf{BLE}, \textbf{Zigbee}, wearables.
    \item \textbf{LPWAN (Low-Power Wide-Area Network)}: Range $\sim$ tens of km. Example: \textbf{LoRaWAN}. Used for agriculture and industrial monitoring.
\end{itemize}

\subsection{12.1 PAN Technology: Bluetooth Low Energy (BLE)}
Designed to replace cables and run on coin cell batteries, use in low energy mode (LE).
\begin{itemize}
    \item \textbf{Communication Roles}: \textbf{Advertiser}: Broadcasts packets (e.g., headphone). \textbf{Scanner}: Receives packets without connecting. \textbf{Initiator}: Attempts to form connection (becomes \textbf{Master}).
    \item \textbf{Channels}: 40 channels total (2 MHz wide).
    \begin{itemize}
        \item \textbf{37, 38, 39}: Advertising channels (placed to avoid WiFi interference).
        \item \textbf{0--36}: Data channels using \textbf{FHSS} (Frequency Hopping Spread Spectrum).
    \end{itemize}
    \item \textbf{The Connection Process}
  \begin{enumerate}
    \item \textbf{Advertising Event}: \textit{Advertiser} broadcasts packets on advertising channels (periodic).
    \item \textbf{Initiation}: A \textit{Scanner} receives the advertisement. If it wants to connect, it becomes an \textbf{Initiator} and sends a connection request on the same channel.
    \item \textbf{Piconet Formation}: If the Advertiser accepts, a \textbf{Piconet} (point-to-point connection) is formed. Initiator becomes \textbf{Master}. Advertiser becomes \textbf{Slave}.
    \item \textbf{Data Exchange}: Once connected, pair moves to data channels (0-36). Use \textbf{FHSS} to switch channels after connection events (typically every 4.615 ms) to combat interference, fading.
  \end{enumerate}
  \item \textbf{Types of BLE Events}: \textbf{Advertising}: Signal wish to pair or relay broadcast message (beacons). \textbf{Periodic Advertising}: device advertises over non-primary channels using hopping pattern. (Intervals can range \textbf{7.5 ms to 81.9 s}). \textbf{Connecting}: handshake pairing device to host.
\end{itemize}

\subsection{12.2 PAN Technology: ZigBee}
Based on IEEE 802.15.4, operating at 2.4 GHz using DSSS.
\begin{itemize}
    \item \textbf{Mesh Topology}: Unlike BLE (Star), Zigbee supports \textbf{many-to-many} mesh networking, allowing data to hop through routers.
    \item \textbf{Network Components}:
    \begin{itemize}
        \item \textbf{ZigBee Controller (ZC)}: One per network; initiates network functions (assign logical addresses, permits node to join or leave mesh).
        \item \textbf{ZigBeeRouter (ZR)}: Optional; handles multi-hop routing and joins nodes, handles some of the load of mesh network hopping and routing coordination.
        \item \textbf{End Device (ZED)}: Simple sensor; no routing logic (e.g. light switch).
    \end{itemize}
\end{itemize}

\subsection{12.3 Low-Power Wide-Area Network LPWAN: LoRaWAN}
Long-range Wide-Area Network LoRaWAN: most popular LPWAN protocol.
\begin{itemize}
    \item \textbf{Physical Layer (LoRa)}: Uses \textbf{Chirp Spread Spectrum (CSS)}.
    \begin{itemize}
      \item \textbf{Chirp}: Signal frequency increases linearly over time. (like chirp)
      \item \textbf{Spreading Factor (SF)}: Controls symbol duration. Spreads signal across wide frequency band. Higher SF = longer range, robust to noise, multipath fading but \textbf{lower data rate}.
      \item \textbf{Encoding}: $n$ bits encoded by dividing bandwidth into $2^n$ possible initial frequencies.
      \item \textbf{Orthogonal}: Diff SFs do not interfere, simultaneous transmissions on same frequency.
    \end{itemize}
    \item \textbf{MAC Layer}: 
    \begin{itemize}
      \item \textbf{Star Topology}: End-nodes $\rightarrow$ Gateway $\rightarrow$ Backhaul (4G/WiFi) $\rightarrow$ Cloud Server.
      \item \textbf{Channels}: \textbf{Uplink}: 64 channels (125 kHz) and 8 channels (500 kHz) using frequency hopping. \textbf{Downlink}: 8 channels (500 kHz).
      \item \textbf{ALOHA Access}: Devices transmit whenever they have data; no carrier sensing.
      \item \textbf{Diversity}: Rely on random channel selection (63 uplink), orthogonality to min. collisions.
    \end{itemize}
    \item \textbf{Constraints \& Weaknesses}: high \textbf{Latency}: (40ms--1.2s per message); no real-time capability. \textbf{Mobility}: Challenging for fast-moving nodes.\textbf{Asymmetry}: Primarily one-way (uplink); downlink is severely limited. \textbf{Dependency}: Lacks full OSI stack; requires a cloud-based interface/subscription.
\end{itemize}

\section{13. Cellular Networks}

\textbf{`Cell'}: Instead of a high-power tower, area divided into smaller Hexagonal Cells.
\begin{itemize}
    \item \textbf{Benefit}: Phones can transmit at much \textbf{lower power} because the tower is closer.
    \item \textbf{Frequency Reuse}: Neighboring cells use diff freq (channels) to prevent interference.
    \item \textbf{Generational Evolution of Cellular Networks}:
\end{itemize}

\subsubsection{1G (Analog) \& 2G (Digital)}
\begin{itemize}
    \item \textbf{1G (AMPS)} - Advanced mobile phone services: Analog voice, frequency modulation.
    \item \textbf{2G (GSM)} - Global System for Mobile Comms: Shift to Digital.
    \begin{itemize}
        \item \textbf{Logic}: Combined \textbf{TDMA} (Time Slots) with slow frequency hopping to fight fading.
        \item \textbf{Features}: Introduced SIM (Subscriber Identity Module) cards, SMS, encryption.
    \end{itemize}
    \item \textbf{2.5G (GPRS)} - General Packet Radio Service: \textbf{Packet Switching} for bursty internet data.
\end{itemize}

\subsubsection{3G (CDMA - Code Division Multiple Access), 4G (LTE - Long Term Evolution)}
\begin{itemize}
    \item \textbf{CDMA Principle}: Everyone talks at the same time on the same frequency, but each user uses a unique \textbf{Orthogonal Code}. (Application of DSSS for multiple access). \textbf{Math}: If Code A $\cdot$ Code B = 0, the receiver can "filter out" the unwanted user.
    \item \textbf{LTE Architecture}: Split into \textbf{RAN} (Radio Access Network / eNodeB) and \textbf{Core Network}. Uses \textbf{OFDMA} and \textbf{MIMO}. Completely IP-based, packet-switched network.
    \item \textbf{LTE-Advanced}: Introduced Carrier Aggregation (combining channels for up to 100 MHz).
\end{itemize}

\subsubsection{5G (New Radio)}
\begin{itemize}
    \item \textbf{Massive MIMO}: Using arrays with $>64$ antennas for fine-grained beamforming.
    \item \textbf{Terahertz/mmWave}: Moving to much higher frequencies for Gbps speeds.
\end{itemize}
